local function Create(className, parent, properties)
    local instance = Instance.new(className)
    if properties then
        for prop, value in pairs(properties) do
            instance[prop] = value
        end
    end
    instance.Parent = parent
    return instance
end

local function AddCorner(instance, radius)
    local corner = Instance.new("UICorner", instance)
    corner.CornerRadius = radius or UDim.new(0, 12)
    return corner
end

local function AddStroke(instance, color, thickness)
    local stroke = Instance.new("UIStroke", instance)
    stroke.Color = color or Configs.Colors.Stroke
    stroke.Thickness = thickness or 1.5
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return stroke
end

local function Tween(instance, properties, duration, easing)
    local tween = TweenService:Create(instance, 
        TweenInfo.new(duration or 0.3, easing or Enum.EasingStyle.Quad), 
        properties
    )
    tween:Play()
    return tween
end

local function SafeCall(func, ...)
    local success, result = pcall(func, ...)
    return success, result
end

local function TableClone(tbl)
    local new = {}
    for k, v in pairs(tbl) do
        if type(v) == "table" then
            new[k] = TableClone(v)
        else
            new[k] = v
        end
    end
    return new
end

local function RandomColor()
    return Color3.fromRGB(math.random(0, 255), math.random(0, 255), math.random(0, 255))
end

local function GetExploitName()
    local exploit = "Unknown"
    SafeCall(function()
        if identifyexecutor then exploit = identifyexecutor()
        elseif getexecutorname then exploit = getexecutorname()
        elseif syn and syn.getexecutorname then exploit = syn.getexecutorname()
        end
    end)
    return exploit
end

local function Clipboard(text)
    SafeCall(function()
        if setclipboard then setclipboard(text)
        elseif syn and syn.setclipboard then syn.setclipboard(text)
        elseif toclipboard then toclipboard(text)
        end
    end)
end

function SoundManager:Play(soundId)
    if not self.Enabled then return end
    local sound = Instance.new("Sound")
    sound.SoundId = soundId or "rbxassetid://88442833509532"
    sound.Volume = self.Volume
    sound.Parent = CoreGui
    sound:Play()
    table.insert(self.Sounds, sound)
    task.spawn(function()
        task.wait(sound.TimeLength or 5)
        sound:Destroy()
    end)
end

function SoundManager:SetVolume(volume)
    self.Volume = math.clamp(volume, 0, 1)
end

function SoundManager:Toggle()
    self.Enabled = not self.Enabled
end

local Bypass = {
    Active = false,
    Remotes = {},
    Hooked = {}
}

function Bypass:FindRemotes()
    local remotes = {}
    local containers = {
        RepStorage,
        Workspace,
        Lighting,
        CoreGui,
        LP:FindFirstChild("PlayerGui"),
        LP:FindFirstChild("Backpack"),
        LP:FindFirstChild("PlayerScripts"),
        LP:FindFirstChild("Character") or Workspace:FindFirstChild(LP.Name)
    }
    
    for _, container in ipairs(containers) do
        if container then
            SafeCall(function()
                for _, obj in ipairs(container:GetDescendants()) do
                    if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                        local name = obj.Name:lower()
                        if name:match("ace") or name:match("anticheat") or name:match("anti") or name:match("cheat") then
                            table.insert(remotes, obj)
                        end
                    end
                end
            end)
        end
    end
    
    return remotes
end

function Bypass:Apply()
    if self.Active then
        Notify({Title = "🛡️ Bypass", Text = "الحماية مكسورة مسبقاً", Type = "warning"})
        return
    end
    
    local remotes = self:FindRemotes()
    local count = 0
    
    for _, remote in ipairs(remotes) do
        SafeCall(function()
            if remote:IsA("RemoteEvent") and remote.FireServer then
                local old = remote.FireServer
                self.Hooked[remote] = hookfunction(old, function(...) return end)
                count = count + 1
            elseif remote:IsA("RemoteFunction") and remote.InvokeServer then
                local old = remote.InvokeServer
                self.Hooked[remote] = hookfunction(old, function(...) return true end)
                count = count + 1
            end
        end)
    end
    
    -- Disable logs
    SafeCall(function()
        if getconnections then
            local conns = getconnections(LogService.MessageOut)
            for _, conn in ipairs(conns) do
                conn:Disconnect()
            end
        end
    end)
    
    self.Active = true
    Notify({
        Title = "🛡️ Bypass",
        Text = "تم فك الحماية بنجاح! (" .. count .. " ريموت)",
        Type = "success",
        Icon = "🔓"
    })
end

function Bypass:Restore()
    for remote, hook in pairs(self.Hooked) do
        SafeCall(function()
            if remote:IsA("RemoteEvent") then
                remote.FireServer = hook
            elseif remote:IsA("RemoteFunction") then
                remote.InvokeServer = hook
            end
        end)
    end
    self.Hooked = {}
    self.Active = false
end

local Flight = {
    Active = false,
    Speed = 60,
    MaxSpeed = 250,
    MinSpeed = 10,
    Connections = {},
    Parts = {},
    Mode = "free", -- free, hover, follow
    Target = nil
}

function Flight:Start()
    if self.Active then return end
    local char = LP.Character
    if not char then
        Notify({Title = "✈️ Flight", Text = "لا توجد شخصية", Type = "warning"})
        return
    end
    
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then
        Notify({Title = "✈️ Flight", Text = "تعذر العثور على الأجزاء", Type = "danger"})
        return
    end
    
    self:Stop()
    
    local gyro = Instance.new("BodyGyro", root)
    gyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    gyro.P = 9e4
    gyro.D = 9e3
    gyro.CFrame = root.CFrame
    
    local vel = Instance.new("BodyVelocity", root)
    vel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    vel.Velocity = Vector3.zero
    
    hum.PlatformStand = true
    self.Active = true
    self.Parts = {Gyro = gyro, Velocity = vel}
    
    local camera = Workspace.CurrentCamera
    
    local connection = RunService.Heartbeat:Connect(function()
        if not self.Active or not root or not hum or hum.Health <= 0 then
            self:Stop()
            return
        end
        
        gyro.CFrame = camera.CFrame
        
        local moveDir = hum.MoveDirection
        if moveDir.Magnitude > 0.05 then
            local camCFrame = camera.CFrame
            local forward = camCFrame.LookVector
            local right = camCFrame.RightVector
            local up = camCFrame.UpVector
            
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                vel.Velocity = forward * self.Speed * 2 + up * self.Speed * 0.5
            elseif UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                vel.Velocity = forward * self.Speed * 0.5 - up * self.Speed * 0.5
            else
                local relative = camCFrame:VectorToObjectSpace(moveDir)
                vel.Velocity = (forward * -relative.Z + right * relative.X + up * relative.Y) * self.Speed
            end
        else
            if self.Mode == "hover" then
                vel.Velocity = Vector3.new(0, 0, 0)
            else
                vel.Velocity = Vector3.zero
            end
        end
    end)
    
    self.Connections = {Heartbeat = connection}
    
    Notify({
        Title = "✈️ Flight",
        Text = "تم تشغيل الطيران | السرعة: " .. self.Speed,
        Type = "success",
        Icon = "🛩️"
    })
end

    for _, conn in pairs(self.Connections) do

    end
    self.Connections = {}
