local function Create(className, parent, properties)
    local object = Instance.new(className)

    if properties then
        for property, value in pairs(properties) do
            pcall(function()
                object[property] = value
            end)
        end
    end

    object.Parent = parent
    return object
end

local function Tween(object, time, properties, style, direction)
    if not object then
        return
    end

    local info = TweenInfo.new(
        time or 0.2,
        style or Enum.EasingStyle.Quint,
        direction or Enum.EasingDirection.Out
    )

    local animation = TweenService:Create(object, info, properties)
    animation:Play()

    return animation
end

local function Round(object, radius)
    return Create("UICorner", object, {
        CornerRadius = UDim.new(0, radius or CFG.Radius)
    })
end

local function Outline(object, color, thickness, transparency)
    return Create("UIStroke", object, {
        Color = color or Color3.fromRGB(60, 60, 72),
        Thickness = thickness or 1,
        Transparency = transparency or 0
    })
end

local function AddPadding(object, left, right, top, bottom)
    return Create("UIPadding", object, {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
    })
end

local function GetCharacter()
    local char = LocalPlayer.Character

    if not char then
        return nil
    end

    return char
end

local function GetHumanoid()
    local char = GetCharacter()

    if not char then
        return nil
    end

    return char:FindFirstChildOfClass("Humanoid")
end

local function GetRoot()
    local char = GetCharacter()

    if not char then
        return nil
    end

    return char:FindFirstChild("HumanoidRootPart")
        or char:FindFirstChild("Torso")
end

local function AddPage(name, subtitle, icon)
    local page = Create("Frame", PageContainer, {
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        BackgroundTransparency = 1,
        Visible = false
    })

    local layout = Create("UIListLayout", page, {
        Padding = UDim.new(0, 10)
    })

    local button, indicator, iconLabel, label = CreateTabButton(name, icon)

    local tab = {
        Name = name,
        Subtitle = subtitle,
        Icon = icon,
        Page = page,
        Button = button,
        Indicator = indicator,
        IconLabel = iconLabel,
        Label = label,
    }

    table.insert(Tabs, tab)

    button.MouseButton1Click:Connect(function()
        for _, other in ipairs(Tabs) do
            other.Page.Visible = false

            Tween(other.Button, 0.18, {
                BackgroundTransparency = 1
            })

            Tween(other.Indicator, 0.18, {
                BackgroundTransparency = 1
            })

            Tween(other.Label, 0.18, {
                TextColor3 = CFG.Gray
            })

            Tween(other.IconLabel, 0.18, {
                TextColor3 = CFG.Gray
            })
        end

        page.Visible = true

        Tween(button, 0.18, {
            BackgroundColor3 = CFG.CardColor,
            BackgroundTransparency = 0
        })

        Tween(indicator, 0.18, {
            BackgroundTransparency = 0
        })

        Tween(label, 0.18, {
            TextColor3 = CFG.White
        })

        Tween(iconLabel, 0.18, {
            TextColor3 = CFG.Accent
        })

        PageTitle.Text = name
        PageSubtitle.Text = subtitle

        ActiveTab = tab
    end)

    return tab
end

local function AddToggle(parent, text, default, callback)
    local enabled = default or false

    local button = Create("TextButton", parent, {
        Size = UDim2.new(1, 0, 0, 46),
        BackgroundColor3 = CFG.CardColor,
        AutoButtonColor = false,
        Text = ""
    })

    Round(button, 11)
    Outline(button, Color3.fromRGB(48, 48, 60), 1)

    local label = Create("TextLabel", button, {
        Size = UDim2.new(1, -85, 1, 0),
        Position = UDim2.new(0, 14, 0, 0),
        BackgroundTransparency = 1,
        Text = text,
        Font = Enum.Font.GothamMedium,
        TextSize = 12,
        TextColor3 = CFG.White,
        TextXAlignment = Enum.TextXAlignment.Left
    })

    local toggle = Create("Frame", button, {
        Size = UDim2.new(0, 48, 0, 26),
        Position = UDim2.new(1, -60, 0.5, -13),
        BackgroundColor3 = Color3.fromRGB(45, 45, 55)
    })

    Round(toggle, 20)

    local knob = Create("Frame", toggle, {
        Size = UDim2.new(0, 20, 0, 20),
        Position = UDim2.new(0, 3, 0.5, -10),
        BackgroundColor3 = CFG.White
    })

    Round(knob, 20)

    local function Refresh()
        if enabled then
            Tween(toggle, 0.2, {
                BackgroundColor3 = CFG.Accent
            })

            Tween(knob, 0.2, {
                Position = UDim2.new(1, -23, 0.5, -10)
            })
        else
            Tween(toggle, 0.2, {
                BackgroundColor3 = Color3.fromRGB(45, 45, 55)
            })

            Tween(knob, 0.2, {
                Position = UDim2.new(0, 3, 0.5, -10)
            })
        end
    end

    button.MouseButton1Click:Connect(function()
        enabled = not enabled
        Refresh()
        pcall(callback, enabled)
    end)

    Refresh()

    return {
        Get = function()
            return enabled
        end,

        Set = function(value)
            enabled = value
            Refresh()
            pcall(callback, enabled)
        end,

        Button = button
    }
end

local Flying = false
local FlySpeed = 50

local FlightConnection = nil
local BodyVelocity = nil
local BodyGyro = nil
local FlightRoot = nil
local FlightHumanoid = nil

function StopFly()
    if FlightConnection then
        FlightConnection:Disconnect()
        FlightConnection = nil
    end

    if BodyVelocity then
        BodyVelocity:Destroy()
        BodyVelocity = nil
    end

    if BodyGyro then
        BodyGyro:Destroy()
        BodyGyro = nil
    end

    if FlightHumanoid then
        FlightHumanoid.PlatformStand = false
        FlightHumanoid = nil
    end

    Flying = false
    FlightRoot = nil

    Notify(
        "Fly",
        "تم إيقاف الطيران",
        2,
        "default"
    )
end

function StartFly()
    if Flying then
        Notify(
            "Fly",
            "الطيران يعمل مسبقاً",
            2,
            "warning"
        )
        return
    end

    local character = GetCharacter()
    local humanoid = GetHumanoid()
    local root = GetRoot()

    if not character or not humanoid or not root then
        Notify(
            "Fly",
            "تعذر العثور على الشخصية",
            2,
            "error"
        )
        return
    end

    Flying = true
    FlightRoot = root
    FlightHumanoid = humanoid

    BodyGyro = Instance.new("BodyGyro")
    BodyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    BodyGyro.P = 40000
    BodyGyro.CFrame = root.CFrame
    BodyGyro.Parent = root

    BodyVelocity = Instance.new("BodyVelocity")
    BodyVelocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    BodyVelocity.Velocity = Vector3.zero
    BodyVelocity.Parent = root

    humanoid.PlatformStand = true

    FlightConnection = RunService.Heartbeat:Connect(function()
        if not Flying then
            return
        end

        if not FlightRoot or not FlightRoot.Parent then
            StopFly()
            return
        end

        local camera = workspace.CurrentCamera

        BodyGyro.CFrame = camera.CFrame

        local direction = Vector3.zero
        local move = FlightHumanoid.MoveDirection

        if move.Magnitude > 0.01 then
            direction = move.Unit
        end

        if direction.Magnitude > 0 then
            BodyVelocity.Velocity = direction * FlySpeed
        else
            BodyVelocity.Velocity = Vector3.zero
        end
    end)

    Notify(
        "Fly",
        "تم تشغيل الطيران",
        2,
        "success"
    )
end

AddButton(
    PlayerPage.Page,
    "زيادة سرعة الطيران +50",
    function()
        FlySpeed += 50

        Notify(
            "Fly Speed",
            "السرعة الحالية: " .. FlySpeed,
            2,
            "success"
        )
    end
)

AddButton(
    PlayerPage.Page,
    "تقليل سرعة الطيران -50",
    function()
        FlySpeed = math.max(50, FlySpeed - 50)

        Notify(
            "Fly Speed",
            "السرعة الحالية: " .. FlySpeed,
            2
        )
    end
)

local ESPEnabled = false

local function RemoveESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local highlight = player.Character:FindFirstChild("STMC_ESP")

            if highlight then
                highlight:Destroy()
            end
        end
    end
end

local function ApplyESP()
    if not ESPEnabled then
        return
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local highlight = player.Character:FindFirstChild("STMC_ESP")

            if not highlight then
                highlight = Instance.new("Highlight")
                highlight.Name = "STMC_ESP"
                highlight.FillColor = CFG.Accent
                highlight.OutlineColor = CFG.White
                highlight.FillTransparency = 0.55
                highlight.OutlineTransparency = 0
                highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                highlight.Parent = player.Character
            end
        end
    end
end

local function RefreshPlayerList()
    if not PlayerButtonsContainer then
        return
    end

    for _, child in ipairs(PlayerButtonsContainer:GetChildren()) do
        if child:IsA("TextButton") or child:IsA("TextLabel") then
            child:Destroy()
        end
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local button = Create("TextButton", PlayerButtonsContainer, {
                Size = UDim2.new(1, 0, 0, 42),
                BackgroundColor3 = CFG.CardColor,
                Text = player.DisplayName .. "  @" .. player.Name,
                Font = Enum.Font.GothamMedium,
                TextSize = 12,
                TextColor3 = CFG.White,
                AutoButtonColor = false
            })

            Round(button, 10)

            button.MouseButton1Click:Connect(function()
                SelectedPlayer = player

                for _, other in ipairs(PlayerButtonsContainer:GetChildren()) do
                    if other:IsA("TextButton") then
                        Tween(other, 0.15, {
                            BackgroundColor3 = CFG.CardColor
                        })
                    end
                end

                Tween(button, 0.15, {
                    BackgroundColor3 = Color3.fromRGB(35, 45, 75)
                })

                Notify(
                    "Player",
                    "تم اختيار " .. player.Name,
                    2,
                    "success"
                )
            end)
        end
    end

    if #Players:GetPlayers() <= 1 then
        Create("TextLabel", PlayerButtonsContainer, {
            Size = UDim2.new(1, 0, 0, 42),
            BackgroundTransparency = 1,
            Text = "لا يوجد لاعبين آخرين",
            Font = Enum.Font.Gotham,
            TextSize = 12,
            TextColor3 = CFG.Gray
        })
    end
end

local dragging = false
local dragStart
local startPos

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = Main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - dragStart

    Main.Position = UDim2.new(
        startPos.X.Scale,
        startPos.X.Offset + delta.X,
        startPos.Y.Scale,
        startPos.Y.Offset + delta.Y
    )

    Shadow.Position = Main.Position + UDim2.fromOffset(0, 10)
end)

local function Responsive()
    local camera = workspace.CurrentCamera

    if not camera then
        return
    end

    local viewport = camera.ViewportSize

    if viewport.X < 760 then
        Main.Size = UDim2.new(
            0,
            math.min(680, viewport.X - 20),
            0,
            math.min(500, viewport.Y - 30)
        )

        Main.Position = UDim2.new(
            0.5,
            -Main.AbsoluteSize.X / 2,
            0.5,
            -Main.AbsoluteSize.Y / 2
        )
    else
        Main.Size = OriginalSize
    end

    Shadow.Size = Main.Size
    Shadow.Position = Main.Position + UDim2.fromOffset(0, 10)
end

workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(Responsive)
