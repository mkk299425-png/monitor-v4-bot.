RunService.Stepped:Connect(function()
    if LP.Character and LP.Character:FindFirstChild("Humanoid") then
        if SpeedEnabled then LP.Character.Humanoid.WalkSpeed = 100 end
        if JumpEnabled then LP.Character.Humanoid.JumpPower = 150 end
    end
    if NoclipEnabled and LP.Character then
        for _, v in pairs(LP.Character:GetDescendants()) do
            if v:IsA("BasePart") then v.CanCollide = false end
        end
    end
end)

UserInputService.JumpRequest:Connect(function()
    if InfJumpEnabled and LP.Character and LP.Character:FindFirstChild("Humanoid") then
        LP.Character.Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

local ESP_Active = false
AddButton(ESPTab, { Name = "👁️ تفعيل كشف اللاعبين (ESP)", Callback = function() ESP_Active = true MakeNotifi({Title="ESP", Text="تم تفعيل الرادار", Time=2}) end })
AddButton(ESPTab, { Name = "❌ إيقاف كشف اللاعبين (ESP)", Callback = function() ESP_Active = false MakeNotifi({Title="ESP", Text="تم إيقاف الرادار", Time=2}) end })

RunService.Heartbeat:Connect(function()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            local highlight = player.Character:FindFirstChild("STMC_ESP")
            if ESP_Active then
                if not highlight then
                    highlight = Instance.new("Highlight")
                    highlight.Name = "STMC_ESP"
                    highlight.FillColor = Color3.fromRGB(255, 0, 0)
                    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                    highlight.FillTransparency = 0.5
                    highlight.OutlineTransparency = 0
                    highlight.Parent = player.Character
                end
            else
                if highlight then highlight:Destroy() end
            end
        end
    end
end)

            end
        end
    end
})

-- Update Dropdowns for Players
local function UpdatePlayerLists()
    local list = {}
    for _, p in pairs(Players:GetPlayers()) do if p ~= LP then table.insert(list, p.Name) end end
    if #list == 0 then list = {"لا يوجد لاعبين"} end
    TPDropdown:Update(list)
    TrollDropdown:Update(list)
end
Players.PlayerAdded:Connect(UpdatePlayerLists); Players.PlayerRemoved:Connect(UpdatePlayerLists); UpdatePlayerLists()
