local function Create(instance, parent, props)
    local new = Instance.new(instance)
    if props then
        for prop, value in pairs(props) do
            new[prop] = value
        end
    end
    new.Parent = parent
    return new
end

local function SetProps(instance, props)
    if instance and props then
        for prop, value in pairs(props) do
            instance[prop] = value
        end
    end
    return instance
end

local function Corner(parent, props)
    local new = Create("UICorner", parent)
    new.CornerRadius = Configs_HUB.Corner_Radius
    if props then SetProps(new, props) end
    return new
end

local function Stroke(parent, props)
    local new = Create("UIStroke", parent)
    new.Color = Configs_HUB.Cor_Stroke
    new.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    new.Thickness = 2
    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.25, Color3.fromRGB(180, 180, 180)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(15, 15, 15)),
        ColorSequenceKeypoint.new(0.75, Color3.fromRGB(90, 90, 90)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 255, 255))
    }
    gradient.Rotation = 0
    gradient.Parent = new
    coroutine.wrap(function()
        while new.Parent do
            gradient.Rotation = (gradient.Rotation + 2) % 360
            RunService.RenderStepped:Wait()
        end
    end)()
    if props then SetProps(new, props) end
    return new
end

local function CreateTween(instance, prop, value, time, tweenWait)
    local tween = TweenService:Create(instance,
    TweenInfo.new(time, Enum.EasingStyle.Linear),
    {[prop] = value})
    tween:Play()
    if tweenWait then tween.Completed:Wait() end
end

local function TextSetColor(instance)

    end)
    instance.MouseLeave:Connect(function()
        CreateTween(instance, "TextColor3", Configs_HUB.Cor_Text, 0.4, false)
    end)
end

local function SafeHookFunction(func, newfunc)
    if hookfunction then
        return hookfunction(func, newfunc)
    end
    return func
end

local function FindAllACERemotes()
    local remotes = {}
    local containers = {
        RepStorage,
        workspace,
        game:GetService("Lighting"),
        game:GetService("CoreGui"),
        game:GetService("StarterGui"),
        LP:FindFirstChild("PlayerGui"),
        LP:FindFirstChild("Backpack"),
        LP:FindFirstChild("PlayerScripts"),
        LP:FindFirstChild("Character") or workspace:FindFirstChild(LP.Name)
    }
    for _, service in ipairs(game:GetChildren()) do
        if service:IsA("Service") and not table.find(containers, service) then
            table.insert(containers, service)
        end
    end
    for _, container in ipairs(containers) do
        if not container then continue end
        local success, descendants = pcall(function() return container:GetDescendants() end)
        if not success then continue end
        for _, obj in ipairs(descendants) do
            if (obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction")) then
                local name = obj.Name
                if string.find(name, "ACE") or string.find(name:lower(), "ace") then
                    table.insert(remotes, obj)
                end
            end
        end
    end
    return remotes
end

local function ApplyBypassToRemote(remote)
    if not remote then return end
    if remote:IsA("RemoteEvent") then
        local oldFire = remote.FireServer
        if oldFire then
            SafeHookFunction(oldFire, function(self, ...)
                local args = {...}
                local msg = tostring(args[1] or "")
                if msg == "HB" then
                    return oldFire(self, unpack(args))
                end
                return nil
            end)
        end
    elseif remote:IsA("RemoteFunction") then
        local oldInvoke = remote.InvokeServer
        if oldInvoke then
            SafeHookFunction(oldInvoke, function(self, ...)
                local args = {...}
                if tostring(args[1]) == "HB" then return "OK" end
                return true
            end)
        end
    end
end

local function DisableSprintMovementScans()
    local ps = LP:FindFirstChild("PlayerScripts")
    if not ps then return end
    local sm = ps:FindFirstChild("SprintMovement")
    if not sm then return end
    if not getgc then return end
    local GC = getgc(true)
    for _, obj in ipairs(GC) do
        if type(obj) == "function" then
            local info = debug.getinfo(obj)
            if info and info.what == "Lua" and info.source and string.find(info.source, "SprintMovement") then
                if info.name == "scanForMovementObjects" or info.name == "scanForGuiObjects" then
                    SafeHookFunction(obj, function() return false, nil end)
                end
            end
        end
    end
end

    end)
end

-- Connection is created after the flight functions below.
-- This button is a true toggle: fly <-> land.
local function stopFly()
    if not Flying then
        FlyPanel.Visible = FlyPanelCreated
        UpdateMainFlyButton()
        return
    end

    Flying = false

    if flightConnection then
        flightConnection:Disconnect()
        flightConnection = nil
    end

    if bodyVelocity then
        bodyVelocity.Velocity = Vector3.zero
        bodyVelocity:Destroy()
        bodyVelocity = nil
    end

    if bodyGyro then
        bodyGyro:Destroy()
        bodyGyro = nil
    end

    if currentHumanoid then
        currentHumanoid.PlatformStand = false
        currentHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        currentHumanoid = nil
    end

    currentRoot = nil
    -- Keep the small flight panel open after landing.
    FlyPanel.Visible = true
    FlyPanelCreated = true
    FlyStopButton.Text = "🛩️ تشغيل الطيران"
    UpdateMainFlyButton()
    MakeNotifi({Title = "Fly", Text = "تم الهبوط — اضغط نفس الزر للطيران مرة ثانية", Time = 2})
end

local function startFly()
    if Flying then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    stopFly()
    currentRoot = root
    currentHumanoid = humanoid

    bodyGyro = Instance.new("BodyGyro", root)
    bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.P = 9e4
    bodyGyro.CFrame = root.CFrame

    bodyVelocity = Instance.new("BodyVelocity", root)
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Velocity = Vector3.zero

    humanoid.PlatformStand = true
    Flying = true
    FlyPanelCreated = true
    FlyPanel.Visible = true
    FlyStopButton.Text = "🛬 هبوط"
    UpdateMainFlyButton()
    UpdateFlyPanel()

    local camera = workspace.CurrentCamera

    flightConnection = RunService.Heartbeat:Connect(function()
        if not Flying or not currentRoot or not currentHumanoid or currentHumanoid.Health <= 0 then
            stopFly()
            return
        end

        bodyGyro.CFrame = camera.CFrame

        local moveDir = currentHumanoid.MoveDirection
        if moveDir.Magnitude > 0.05 then
            local camCFrame = camera.CFrame
            local forward = camCFrame.LookVector
            local right = camCFrame.RightVector
            
            local relativeMove = camCFrame:VectorToObjectSpace(moveDir)
            local finalVel = (forward * -relativeMove.Z) + (right * relativeMove.X)
            
            bodyVelocity.Velocity = finalVel * FlySpeed
        else
            bodyVelocity.Velocity = Vector3.zero
        end
    end)

    MakeNotifi({Title = "Fly", Text = "تم تشغيل الطيران", Time = 2})
end

function AddDropdown(parent, Configs)
    local DropdownName = Configs.Name or "Dropdown!!"
    local Options = Configs.Options or {"1", "2", "3"}
    local Default = Configs.Default or Options[1]
    local Callback = Configs.Callback or function() end
    local Searchable = Configs.Searchable or false -- If true, adds a search TextBox inside dropdown

        Size = UDim2.new(1, 0, 0, 25),

    })
    TextSetColor(TextLabel)

        Size = UDim2.new(0, 20, 0, 20),

    })

        Size = UDim2.new(1, 0, 0, 0),

        PaddingRight = UDim.new(0, 4),
        PaddingTop = UDim.new(0, 4),
        PaddingBottom = UDim.new(0, 4)

        SearchBox = Create("TextBox", DropdownFrame, {
            Size = UDim2.new(1, 0, 0, 25),
            BackgroundColor3 = Configs_HUB.Cor_Hub,
            TextColor3 = Configs_HUB.Cor_Text,
            PlaceholderText = "ابحث...",
            Font = Configs_HUB.Text_Font,
            TextSize = 12,
            TextXAlignment = Enum.TextXAlignment.Left,
            ClearTextOnFocus = false,
            ZIndex = 25
        })
        Corner(SearchBox)
        Stroke(SearchBox)
        -- Put SearchBox as first element in layout
        SearchBox.LayoutOrder = -1
    end
    
    local Selected = Default
    local expanded = false
    local currentSearchText = ""
    
    local function GetTargetHeight()
        local count = #Options
        local totalH = count * 22 + 8
        if Searchable then
            totalH = totalH + 30 -- extra for search box
        end
        return math.min(totalH, 150)
    end
    
    local function UpdateButtons()
        -- Clear existing buttons (except search box)
        for _, child in ipairs(DropdownFrame:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end
        -- Filter options based on currentSearchText
        local filteredOptions = {}
        if currentSearchText ~= "" then
            for _, option in ipairs(Options) do
                if string.find(string.lower(option), string.lower(currentSearchText), 1, true) then
                    table.insert(filteredOptions, option)
                end
            end
        else
            filteredOptions = Options
        end
        
        for _, option in ipairs(filteredOptions) do
            local btn = Create("TextButton", DropdownFrame, {
                Size = UDim2.new(1, 0, 0, 20),
                BackgroundColor3 = Configs_HUB.Cor_Hub,
                Text = option,
                TextColor3 = Configs_HUB.Cor_Text,
                TextSize = 12,
                Font = Configs_HUB.Text_Font,
                AutoButtonColor = false,
                ZIndex = 21
            })
            Corner(btn)
            btn.MouseButton1Click:Connect(function()
                Selected = option
                TextLabel.Text = DropdownName .. ": " .. option
                Callback(option)
                expanded = false
                currentSearchText = ""
                if SearchBox then SearchBox.Text = "" end
                CreateTween(DropdownFrame, "Size", UDim2.new(1, 0, 0, 0), 0.2, true)
                DropdownFrame.Visible = false
                CreateTween(MainFrame, "Size", UDim2.new(1, 0, 0, 25), 0.2, false)
            end)
        end
        
        -- Update height if expanded
        if expanded then
            local targetH = GetTargetHeight()
            DropdownFrame.Size = UDim2.new(1, 0, 0, targetH)
            MainFrame.Size = UDim2.new(1, 0, 0, 28 + targetH)
        end
    end
    
    UpdateButtons()
    
    -- If search box exists, connect TextChanged
    if SearchBox then
        SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
            currentSearchText = SearchBox.Text
            UpdateButtons()
        end)
    end
    
    TextButton.MouseButton1Click:Connect(function()
        expanded = not expanded
        if expanded then
            DropdownFrame.Visible = true
            local targetH = GetTargetHeight()
            CreateTween(DropdownFrame, "Size", UDim2.new(1, 0, 0, targetH), 0.2, false)
            CreateTween(MainFrame, "Size", UDim2.new(1, 0, 0, 28 + targetH), 0.2, false)
        else
            CreateTween(DropdownFrame, "Size", UDim2.new(1, 0, 0, 0), 0.2, true)
            DropdownFrame.Visible = false
            CreateTween(MainFrame, "Size", UDim2.new(1, 0, 0, 25), 0.2, false)
        end
    end)
    
    return {
        Update = function(self, newOptions)
            Options = newOptions
            UpdateButtons()
        end,
        GetSelected = function(self)
            return Selected
        end
    }
end

    -- Execute command function
    local function ExecuteCommand(cmdText)
        if not cmdText or cmdText == "" then
            MakeNotifi({Title = "تنبيه", Text = "الرجاء كتابة أمر", Time = 2})
            return
        end
        
        -- Check if it's a command (starts with ;)
        if string.sub(cmdText, 1, 1) == ";" then
            -- Remove the ; and parse
            local fullCmd = string.sub(cmdText, 2)
            local parts = {}
            for part in string.gmatch(fullCmd, "[^%s]+") do
                table.insert(parts, part)
            end
            
            local commandName = parts[1]
            local args = {}
            for i = 2, #parts do
                table.insert(args, parts[i])
            end
            
            -- Handle ;titler me <text>
            if commandName == "titler" and args[1] == "me" then
                -- Extract the text (everything after "me")
                local textStart = string.find(fullCmd, "me") + 2
                local text = string.sub(fullCmd, textStart + 1)
                text = string.gsub(text, "^%s+", "") -- Trim leading spaces
                
                if text and text ~= "" then
                    -- Create a Title above player's head
                    local char = LP.Character
                    if char then
                        local head = char:FindFirstChild("Head")
                        if head then
                            -- Remove old title if exists
                            local oldTitle = char:FindFirstChild("STMC_Title")
                            if oldTitle then oldTitle:Destroy() end
                            
                            -- Create BillboardGui
                            local billboard = Instance.new("BillboardGui")
                            billboard.Name = "STMC_Title"
                            billboard.Adornee = head
                            billboard.Size = UDim2.new(0, 300, 0, 50)
                            billboard.StudsOffset = Vector3.new(0, 3.5, 0)
                            billboard.AlwaysOnTop = true
                            billboard.Parent = char
                            
                            -- Title frame
                            local frame = Instance.new("Frame")
                            frame.Size = UDim2.new(1, 0, 1, 0)
                            frame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
                            frame.BackgroundTransparency = 0.5
                            frame.Parent = billboard
                            Corner(frame)
                            
                            -- Text label
                            local label = Instance.new("TextLabel")
                            label.Size = UDim2.new(1, -10, 1, -5)
                            label.Position = UDim2.new(0, 5, 0, 2.5)
                            label.BackgroundTransparency = 1
                            label.Text = text
                            label.TextColor3 = Color3.fromRGB(255, 255, 100)
                            label.TextScaled = true
                            label.Font = Enum.Font.GothamBold
                            label.Parent = frame
                            
                            -- Auto remove after 10 seconds
                            task.delay(10, function()
                                local title = char:FindFirstChild("STMC_Title")
                                if title then title:Destroy() end
                            end)
                            
                            MakeNotifi({Title = "✅ تم التنفيذ", Text = "تم عرض النص فوق رأسك لمدة 10 ثوانٍ", Time = 3})
                        else
                            MakeNotifi({Title = "خطأ", Text = "لا يوجد رأس للشخصية", Time = 2})
                        end
                    else
                        MakeNotifi({Title = "خطأ", Text = "الشخصية غير موجودة", Time = 2})
                    end
                else
                    MakeNotifi({Title = "تنبيه", Text = "الرجاء كتابة النص بعد me", Time = 2})
                end
            else
                MakeNotifi({Title = "تنبيه", Text = "الأمر غير معروف: " .. commandName, Time = 3})
            end
        else
            -- Not a command, send as normal chat
            MakeNotifi({Title = "تنبيه", Text = "الأوامر يجب أن تبدأ بـ ;", Time = 2})
        end
    end
    
    -- Send button click
    SendCmd.MouseButton1Click:Connect(function()
        local cmd = CmdInput.Text
        ExecuteCommand(cmd)
        CmdInput.Text = ""
    end)

    CmdInput.FocusLost:Connect(function(enterPressed)

            CmdInput.Text = ""
        end
    end)

end

local function GetESPColor(player)
    local ok, color = pcall(function()
        if player.Team and player.Team.TeamColor then
            return player.Team.TeamColor.Color
        end
        return Color3.fromRGB(0, 170, 255)
    end)
    return ok and color or Color3.fromRGB(0, 170, 255)
end

local function GetCharacterRoot(character)
    return character and (character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("UpperTorso")
        or character:FindFirstChild("Torso"))
end

local function IsTarget(player)
    return player and ESP_TargetUserId ~= nil and player.UserId == ESP_TargetUserId
end

local function GetOrCreateESPFolder(character)
    if not character then return nil end
    local folder = character:FindFirstChild(ESP_FOLDER_NAME)
    if folder and folder:IsA("Folder") then
        return folder
    end

    if folder then
        pcall(function() folder:Destroy() end)
    end

    folder = Instance.new("Folder")
    folder.Name = ESP_FOLDER_NAME
    folder.Parent = character
    return folder
end

local function RemoveESP(player)
    if not player or not player.Character then return end
    local character = player.Character

local function UpdateNameVisual(player, billboard, distance)
    if not player or not billboard then return end

    local card = billboard:FindFirstChild("Card")
    if not card then return end

    local nameLabel = card:FindFirstChild("PlayerName")
    local infoLabel = card:FindFirstChild("Info")
    local targetBadge = card:FindFirstChild("TargetBadge")
    local humanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")

    local target = IsTarget(player)
    local baseColor = target and Color3.fromRGB(255, 55, 55) or GetESPColor(player)

    if nameLabel then
        nameLabel.Text = player.DisplayName ~= "" and player.DisplayName or player.Name
        nameLabel.TextColor3 = target and Color3.fromRGB(255, 85, 85) or Color3.fromRGB(255, 255, 255)

        local accent = nameLabel:FindFirstChild("NameStroke")
        if accent and accent:IsA("UIStroke") then
            accent.Color = baseColor
        end
    end

    if targetBadge then
        targetBadge.Visible = target
        if target then
            targetBadge.Text = "TARGET"
        end
    end

    if infoLabel then
        local infoParts = {}
        if ESP_ShowHealth and humanoid then
            table.insert(infoParts, "HP " .. math.max(0, math.floor(humanoid.Health + 0.5)))
        end
        if ESP_ShowDistance and distance and distance < math.huge then
            table.insert(infoParts, math.floor(distance + 0.5) .. "m")
        end
        infoLabel.Text = table.concat(infoParts, "  •  ")
        infoLabel.Visible = (#infoParts > 0)
        infoLabel.TextColor3 = target and Color3.fromRGB(255, 180, 180) or Color3.fromRGB(215, 225, 235)
    end
end

-- 🎯 تتبّع المهاجم (بدون Dropdown وبدون Target يدوي)
-- =============================================
local AttackerFollowEnabled = false
local CurrentAttacker = nil
local CurrentAttackerUserId = nil
local AttackerFollowConnection = nil
local AttackerHealthConnection = nil

local FOLLOW_DISTANCE = 3 -- المسافة الثابتة خلف المهاجم

local function GetRoot(character)
    return character and (character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Torso"))
end

local function StopAttackerFollow(silent)
    if AttackerFollowConnection then
        AttackerFollowConnection:Disconnect()
        AttackerFollowConnection = nil
    end
    if AttackerHealthConnection then
        AttackerHealthConnection:Disconnect()
        AttackerHealthConnection = nil
    end
    CurrentAttacker = nil
    CurrentAttackerUserId = nil
    if not silent then
        MakeNotifi({Title="تتبّع المهاجم", Text="⏸️ بانتظار مهاجم...", Time=2})
    end
end

local function StartAttackerFollow(player)
    if not player or player == LP then return end
    if not AttackerFollowEnabled then return end

    CurrentAttacker = player
    CurrentAttackerUserId = player.UserId

    MakeNotifi({
        Title = "🎯 تتبّع المهاجم",
        Text = "تم تحديد المهاجم: " .. player.Name .. " — جاري التتبع من الخلف",
        Time = 4
    })

    -- نراقب موت المهاجم
    if AttackerHealthConnection then AttackerHealthConnection:Disconnect() end
    AttackerHealthConnection = player.CharacterAdded:Connect(function()
        task.wait(0.5)
        if AttackerFollowEnabled and CurrentAttackerUserId == player.UserId then
            CurrentAttacker = player
        end
    end)

    -- نراقب خروج اللاعب
    local function onPlayerRemoving(plr)
        if plr == player and AttackerFollowEnabled then
            StopAttackerFollow(false)
            MakeNotifi({Title="تتبّع المهاجم", Text="خرج المهاجم من السيرفر — بانتظار مهاجم جديد...", Time=3})
        end
    end
    Players.PlayerRemoving:Connect(onPlayerRemoving)

    -- التتبع المستمر
    if AttackerFollowConnection then AttackerFollowConnection:Disconnect() end
    AttackerFollowConnection = RunService.Heartbeat:Connect(function()
        if not AttackerFollowEnabled then
            StopAttackerFollow(true)
            return
        end

        if not CurrentAttacker or not CurrentAttacker.Character then
            -- المهاجم مات أو اختفت شخصيته
            StopAttackerFollow(false)
            return
        end

        local myChar = LP.Character
        local myRoot = GetRoot(myChar)
        local attackerRoot = GetRoot(CurrentAttacker.Character)

        if not myRoot or not attackerRoot then
            return
        end

        -- نحدد موقع خلف المهاجم
        local lookVector = CurrentAttacker.Character:FindFirstChildOfClass("Humanoid") and CurrentAttacker.Character:FindFirstChildOfClass("Humanoid").MoveDirection or Vector3.zero
        local direction = lookVector.Magnitude > 0.1 and lookVector.Unit or attackerRoot.CFrame.LookVector
        local targetPosition = attackerRoot.Position - direction * FOLLOW_DISTANCE

        -- ننقل الشخصية خلف المهاجم
        myRoot.CFrame = CFrame.lookAt(targetPosition, attackerRoot.Position)
    end)
end

-- ربط الضرر: نراقب تغير الـHealth لتحديد المهاجم
local function CheckForAttackerDamage()
    if not AttackerFollowEnabled then return end
    if not LP.Character or not LP.Character:FindFirstChildOfClass("Humanoid") then return end

    local myHumanoid = LP.Character:FindFirstChildOfClass("Humanoid")
    local currentHealth = myHumanoid.Health
    local maxHealth = myHumanoid.MaxHealth

    -- إذا انخفضت الصحة، نبحث عن أقرب لاعب
    if currentHealth < maxHealth then
        local myRoot = GetRoot(LP.Character)
        if not myRoot then return end

        local nearestPlayer = nil
        local nearestDistance = math.huge

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP and player.Character then
                local root = GetRoot(player.Character)
                if root then
                    local distance = (root.Position - myRoot.Position).Magnitude
                    if distance < nearestDistance then
                        nearestDistance = distance
                        nearestPlayer = player
                    end
                end
            end
        end

        if nearestPlayer and nearestDistance < 50 then
            StartAttackerFollow(nearestPlayer)
        end
    end
end

-- مراقبة الصحة لتحديد المهاجم
RunService.Heartbeat:Connect(function()
    if AttackerFollowEnabled then
        CheckForAttackerDamage()
    end
end)

local function CreateShield()
    if LP.Character then
        local shield = LP.Character:FindFirstChild("STMC_Shield")
        if not shield then
            shield = Instance.new("ForceField")
            shield.Name = "STMC_Shield"
            shield.Visible = true
            shield.Parent = LP.Character
        end
    end
end

local function RemoveShield()
    if LP.Character then
        local shield = LP.Character:FindFirstChild("STMC_Shield")
        if shield then
            shield:Destroy()
        end
    end
end

local EscapePosition = Vector3.new(284, 40, 118)

local function StopInvisibilityAndFollow(teleportToSafeSpot)
    isInvisActive = false
    if invisConnection then invisConnection:Disconnect() invisConnection = nil end
    if followConnection then followConnection:Disconnect() followConnection = nil end
    if targetDeathConnection then targetDeathConnection:Disconnect() targetDeathConnection = nil end
    
    local char = LP.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
        
        if teleportToSafeSpot and root then
            root.CFrame = CFrame.new(EscapePosition)
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid.CameraOffset = Vector3.zero end
        workspace.CurrentCamera.CameraSubject = humanoid or root
        
        for _, v in pairs(char:GetDescendants()) do
            if v:IsA("BasePart") and v.Transparency == 0.5 then
                v.Transparency = 0
            end
        end
    end
end

local function UpdatePlayerList()
    local playerNames = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP then
            table.insert(playerNames, player.Name)
        end
    end
    if #playerNames == 0 then table.insert(playerNames, "لا يوجد لاعبين") end
    
    if playerDropdown then pcall(function() playerDropdown:Update(playerNames) end) end
    if TPDropdown then pcall(function() TPDropdown:Update(playerNames) end) end
end
