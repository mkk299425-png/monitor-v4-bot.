function ErrorLogger.Show(message, source)
    if not ErrorLogger.Enabled then return end
    pcall(function()
        CreateErrorPanel()
        if not ErrorLogger.Panel then return end
        
        local timestamp = os.date("%H:%M:%S")
        local fullError = "-----------------------\n"
        fullError = fullError .. "الوقت: " .. timestamp .. "\n"
        fullError = fullError .. "المصدر: " .. tostring(source or "Unknown") .. "\n"
        fullError = fullError .. "-----------------------\n"
        fullError = fullError .. tostring(message) .. "\n\n"
        
        ErrorLogger.CurrentError = fullError
        table.insert(ErrorLogger.Errors, fullError)
        
        if ErrorLogger.ErrorText then
            ErrorLogger.ErrorText.Text = fullError
        end
        if ErrorLogger.Scroll then
            ErrorLogger.Scroll.CanvasPosition = Vector2.new(0, 0)
        end
        if ErrorLogger.Frame then
            ErrorLogger.Frame.Visible = true
        end
        
        warn("[Monetor Error] " .. tostring(message))
    end)
end

local function Create(instance, parent, props)
    local new = Instance.new(instance)
    if props then
        for prop, value in pairs(props) do
            new[prop] = value
        end
    end
    new.Parent = parent
    return new
end

local function SetProps(instance, props)
    if instance and props then
        for prop, value in pairs(props) do
            instance[prop] = value
        end
    end
    return instance
end

local function Corner(parent, props)
    local new = Create("UICorner", parent)
    new.CornerRadius = Configs_HUB.Corner_Radius
    if props then SetProps(new, props) end
    return new
end

local function Stroke(parent, props)
    local new = Create("UIStroke", parent)
    new.Color = Configs_HUB.Cor_Stroke
    new.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    new.Thickness = 2
    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.25, Color3.fromRGB(180, 180, 180)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(15, 15, 15)),
        ColorSequenceKeypoint.new(0.75, Color3.fromRGB(90, 90, 90)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 255, 255))
    }
    gradient.Rotation = 0
    gradient.Parent = new
    coroutine.wrap(function()
        while new.Parent do
            gradient.Rotation = (gradient.Rotation + 2) % 360
            RunService.RenderStepped:Wait()
        end
    end)()
    if props then SetProps(new, props) end
    return new
end

local function CreateTween(instance, prop, value, time, tweenWait)
    local tween = TweenService:Create(instance,
    TweenInfo.new(time, Enum.EasingStyle.Linear),
    {[prop] = value})
    tween:Play()
    if tweenWait then tween.Completed:Wait() end
end

local function TextSetColor(instance)

    end)
    instance.MouseLeave:Connect(function()
        CreateTween(instance, "TextColor3", Configs_HUB.Cor_Text, 0.4, false)
    end)
end

local BypassEnabled = false

local function SafeHookFunction(func, newfunc)
    if hookfunction then
        local ok = pcall(function() hookfunction(func, newfunc) end)
        if ok then return end
    end
    return func
end

local function FindAllACERemotes()
    local remotes = {}
    local containers = {
        RepStorage, workspace,
        Lighting,
        game:GetService("StarterGui"),
        LP:FindFirstChild("PlayerGui"),
        LP:FindFirstChild("Backpack"),
        LP:FindFirstChild("PlayerScripts"),
        LP:FindFirstChild("Character") or workspace:FindFirstChild(LP.Name)
    }
    for _, service in ipairs(game:GetChildren()) do
        if service:IsA("Service") and not table.find(containers, service) then
            table.insert(containers, service)
        end
    end
    for _, container in ipairs(containers) do
        if not container then continue end
        local success, descendants = pcall(function() return container:GetDescendants() end)
        if not success then continue end
        for _, obj in ipairs(descendants) do
            if (obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction")) then
                local name = obj.Name
                if string.find(name, "ACE") or string.find(name:lower(), "ace") then
                    table.insert(remotes, obj)
                end
            end
        end
    end
    return remotes
end

local function ApplyBypassToRemote(remote)
    if not remote then return end
    if remote:IsA("RemoteEvent") then
        local oldFire = remote.FireServer
        if oldFire then
            SafeHookFunction(oldFire, function(self, ...)
                local args = {...}
                local msg = tostring(args[1] or "")
                if msg == "HB" then
                    return oldFire(self, unpack(args))
                end
                return nil
            end)
        end
    elseif remote:IsA("RemoteFunction") then
        local oldInvoke = remote.InvokeServer
        if oldInvoke then
            SafeHookFunction(oldInvoke, function(self, ...)
                local args = {...}
                if tostring(args[1]) == "HB" then return "OK" end
                return true
            end)
        end
    end
end

local function DisableSprintMovementScans()
    pcall(function()
        local ps = LP:FindFirstChild("PlayerScripts")
        if not ps then return end
        local sm = ps:FindFirstChild("SprintMovement")
        if not sm then return end
        if not getgc then return end
        local GC = getgc(true)
        for _, obj in ipairs(GC) do
            if type(obj) == "function" then
                local info = debug.getinfo(obj)
                if info and info.what == "Lua" and info.source and string.find(info.source, "SprintMovement") then
                    if info.name == "scanForMovementObjects" or info.name == "scanForGuiObjects" then
                        SafeHookFunction(obj, function() return false, nil end)
                    end
                end
            end
        end
    end)
end

local function DisableLogs()
    pcall(function()
        if not getconnections then return end
        local conns = getconnections(LogService.MessageOut)
        if conns then
            for _, c in ipairs(conns) do
                pcall(c.Disconnect, c)
            end
        end
    end)
end

local function FullBypass()
    pcall(function()
        if BypassEnabled then
            MakeNotifi({Title = "Monetor X", Text = "الحماية مكسورة مسبقاً", Time = 2})
            return
        end
        local remotes = FindAllACERemotes()
        if #remotes > 0 then
            for _, r in ipairs(remotes) do
                ApplyBypassToRemote(r)
            end
        end
        DisableSprintMovementScans()
        DisableLogs()
        BypassEnabled = true
        MakeNotifi({ Title = "Monetor X", Text = "تم فك الحماية بنجاح", Time = 2 })
    end)
end

local function stopFly()
    if not Flying then
        FlyPanel.Visible = FlyPanelCreated
        UpdateMainFlyButton()
        return
    end

    Flying = false

    if flightConnection then
        flightConnection:Disconnect()
        flightConnection = nil
    end

    if bodyVelocity then
        bodyVelocity.Velocity = Vector3.zero
        bodyVelocity:Destroy()
        bodyVelocity = nil
    end

    if bodyGyro then
        bodyGyro:Destroy()
        bodyGyro = nil
    end

    if currentHumanoid then
        currentHumanoid.PlatformStand = false
        currentHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        currentHumanoid = nil
    end

    currentRoot = nil
    FlyPanel.Visible = true
    FlyPanelCreated = true
    FlyStopButton.Text = "تشغيل الطيران"
    UpdateMainFlyButton()
    MakeNotifi({Title = "Fly", Text = "تم الهبوط", Time = 2})
end

local function startFly()
    if Flying then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    stopFly()
    currentRoot = root
    currentHumanoid = humanoid

    bodyGyro = Instance.new("BodyGyro", root)
    bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.P = 9e4
    bodyGyro.CFrame = root.CFrame

    bodyVelocity = Instance.new("BodyVelocity", root)
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Velocity = Vector3.zero

    humanoid.PlatformStand = true
    Flying = true
    FlyPanelCreated = true
    FlyPanel.Visible = true
    FlyStopButton.Text = "هبوط"
    UpdateMainFlyButton()
    UpdateFlyPanel()

    local camera = workspace.CurrentCamera

    flightConnection = RunService.Heartbeat:Connect(function()
        if not Flying or not currentRoot or not currentHumanoid or currentHumanoid.Health <= 0 then
            stopFly()
            return
        end

        bodyGyro.CFrame = camera.CFrame

        local moveDir = currentHumanoid.MoveDirection
        if moveDir.Magnitude > 0.05 then
            local camCFrame = camera.CFrame
            local forward = camCFrame.LookVector
            local right = camCFrame.RightVector
            
            local relativeMove = camCFrame:VectorToObjectSpace(moveDir)
            local finalVel = (forward * -relativeMove.Z) + (right * relativeMove.X)
            
            bodyVelocity.Velocity = finalVel * FlySpeed
        else
            bodyVelocity.Velocity = Vector3.zero
        end
    end)

    MakeNotifi({Title = "Fly", Text = "تم تشغيل الطيران", Time = 2})
end

FlyStopButton.Activated:Connect(function()
    if Flying then
        stopFly()
    else
        startFly()
    end
end)

LP.CharacterAdded:Connect(function()
    if Flying then stopFly() end
end)

function AddDropdown(parent, Configs)
    local DropdownName = Configs.Name or "Dropdown!!"
    local Options = Configs.Options or {"1", "2", "3"}
    local Default = Configs.Default or Options[1]
    local Callback = Configs.Callback or function() end

        end
    end)()
    Corner(TextButton)
    
    local TextLabel = Create("TextLabel", TextButton, {
        TextSize = 12,
        TextColor3 = Configs_HUB.Cor_Text,
        Text = DropdownName,
        Size = UDim2.new(1, -30, 1, 0),

        TextXAlignment = "Left",

    })

        Image = "rbxassetid://6031090998",
        Size = UDim2.new(0, 20, 0, 20),
        Position = UDim2.new(1, -25, 0.5, 0),
        AnchorPoint = Vector2.new(0, 0.5),
        BackgroundTransparency = 1,
        ImageColor3 = Configs_HUB.Cor_Text
    })
    
    local DropdownFrame = Create("ScrollingFrame", MainFrame, {
        Size = UDim2.new(1, 0, 0, 0),
        Position = UDim2.new(0, 0, 0, 28),
        BackgroundColor3 = Configs_HUB.Cor_Options,
        BackgroundTransparency = 0.1,
        ClipsDescendants = true,
        Visible = false,
        ScrollBarThickness = 3,
        ScrollBarImageColor3 = Configs_HUB.Cor_Stroke,
        CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ZIndex = 20

    Stroke(DropdownFrame)
    
    local DropdownList = Create("UIListLayout", DropdownFrame, {
        Padding = UDim.new(0, 2),
        SortOrder = Enum.SortOrder.LayoutOrder
    })
    
    Create("UIPadding", DropdownFrame, {
        PaddingLeft = UDim.new(0, 4),
        PaddingRight = UDim.new(0, 4),
        PaddingTop = UDim.new(0, 4),
        PaddingBottom = UDim.new(0, 4)
    })
    
    local SearchBox
    if Searchable then
        SearchBox = Create("TextBox", DropdownFrame, {
            Size = UDim2.new(1, 0, 0, 25),
            BackgroundColor3 = Configs_HUB.Cor_Hub,
            TextColor3 = Configs_HUB.Cor_Text,
            PlaceholderText = "ابحث...",
            Font = Configs_HUB.Text_Font,
            TextSize = 12,
            TextXAlignment = Enum.TextXAlignment.Left,
            ClearTextOnFocus = false,
            ZIndex = 25
        })
        Corner(SearchBox)
        Stroke(SearchBox)
        SearchBox.LayoutOrder = -1
    end
    
    local Selected = Default
    local expanded = false
    local currentSearchText = ""
    
    local function GetTargetHeight()
        local count = #Options
        local totalH = count * 22 + 8
        if Searchable then
            totalH = totalH + 30
        end
        return math.min(totalH, 150)
    end
    
    local function UpdateButtons()
        for _, child in ipairs(DropdownFrame:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end
        local filteredOptions = {}
        if currentSearchText ~= "" then
            for _, option in ipairs(Options) do
                if string.find(string.lower(option), string.lower(currentSearchText), 1, true) then
                    table.insert(filteredOptions, option)
                end
            end
        else
            filteredOptions = Options
        end
        
        for _, option in ipairs(filteredOptions) do
            local btn = Create("TextButton", DropdownFrame, {
                Size = UDim2.new(1, 0, 0, 20),
                BackgroundColor3 = Configs_HUB.Cor_Hub,
                Text = option,
                TextColor3 = Configs_HUB.Cor_Text,
                TextSize = 12,
                Font = Configs_HUB.Text_Font,
                AutoButtonColor = false,
                ZIndex = 21
            })
            Corner(btn)
            btn.MouseButton1Click:Connect(function()
                Selected = option
                TextLabel.Text = DropdownName .. ": " .. option
                local ok, err = pcall(Callback, option)
                if not ok then
                    ErrorLogger.Show(err, "Dropdown: " .. DropdownName)
                end
                expanded = false
                currentSearchText = ""
                if SearchBox then SearchBox.Text = "" end
                CreateTween(DropdownFrame, "Size", UDim2.new(1, 0, 0, 0), 0.2, true)
                DropdownFrame.Visible = false
                CreateTween(MainFrame, "Size", UDim2.new(1, 0, 0, 25), 0.2, false)
            end)
        end
        
        if expanded then
            local targetH = GetTargetHeight()
            DropdownFrame.Size = UDim2.new(1, 0, 0, targetH)
            MainFrame.Size = UDim2.new(1, 0, 0, 28 + targetH)
        end
    end
    
    UpdateButtons()
    
    if SearchBox then
        SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
            currentSearchText = SearchBox.Text
            UpdateButtons()
        end)
    end
    
    TextButton.MouseButton1Click:Connect(function()
        expanded = not expanded
        if expanded then
            DropdownFrame.Visible = true
            local targetH = GetTargetHeight()
            CreateTween(DropdownFrame, "Size", UDim2.new(1, 0, 0, targetH), 0.2, false)
            CreateTween(MainFrame, "Size", UDim2.new(1, 0, 0, 28 + targetH), 0.2, false)
        else
            CreateTween(DropdownFrame, "Size", UDim2.new(1, 0, 0, 0), 0.2, true)
            DropdownFrame.Visible = false
            CreateTween(MainFrame, "Size", UDim2.new(1, 0, 0, 25), 0.2, false)
        end
    end)
    
    return {
        Update = function(self, newOptions)
            Options = newOptions
            UpdateButtons()
        end,
        GetSelected = function(self)
            return Selected
        end,
        SetSelected = function(self, value)
            Selected = value
            TextLabel.Text = DropdownName .. ": " .. value
        end
    }
end

                if server.playing < server.maxPlayers and server.id ~= game.JobId then
                    if server.playing > bestPlayers then
                        bestPlayers = server.playing
                        bestServer = server.id
                    end
                end
            end
            
            if bestServer then
                pcall(function()
                    game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId, bestServer, LP)
                end)
            else
                MakeNotifi({Title="Server Hop", Text="ما لقيت سيرفر", Time=3})
            end
        end)
    end
})

local function ApplyHitboxToTarget(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    local char = targetPlayer.Character
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            if not OriginalSizes[part] then OriginalSizes[part] = part.Size end
            part.Size = Vector3.new(HitboxSize, HitboxSize, HitboxSize)
            part.Transparency = 1
            part.CanCollide = false
            part.Massless = true
        end
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root then
        if not OriginalSizes[root] then OriginalSizes[root] = root.Size end
        root.Size = Vector3.new(HitboxSize, HitboxSize, HitboxSize)
        root.Transparency = 1
        root.CanCollide = false
    end
end

local function RestoreHitbox(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    for _, part in ipairs(targetPlayer.Character:GetDescendants()) do
        if part:IsA("BasePart") and OriginalSizes[part] then
            pcall(function()
                part.Size = OriginalSizes[part]
                part.Transparency = 0
                part.CanCollide = true
                part.Massless = false
            end)
            OriginalSizes[part] = nil
        end
    end
end

local TATWIR_Active = false
local TATWIR_SpinSpeed = 200
local TATWIR_BodyAngular = nil
local TATWIR_SpinConnection = nil
local TATWIR_TouchConnection = nil

AddToggleButton(CombatTab, {
    Name = "تطوير",
    OnText = "تطوير: تشغيل",
    OffText = "تطوير: إيقاف",
    StartState = false,
    OnEnable = function()
        local char = LP.Character
        if not char then return end
        local myRoot = char:FindFirstChild("HumanoidRootPart")
        if not myRoot then return end
        TATWIR_Active = true
        
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                for _, part in ipairs(plr.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        pcall(function() part.CanCollide = true end)
                    end
                end
            end
        end
        
        local oldSpin = myRoot:FindFirstChild("Tatwir_Spin")
        if oldSpin then oldSpin:Destroy() end
        
        TATWIR_BodyAngular = Instance.new("BodyAngularVelocity")
        TATWIR_BodyAngular.Name = "Tatwir_Spin"
        TATWIR_BodyAngular.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        TATWIR_BodyAngular.AngularVelocity = Vector3.new(0, math.rad(TATWIR_SpinSpeed), 0)
        TATWIR_BodyAngular.P = 9e9
        TATWIR_BodyAngular.Parent = myRoot
        
        TATWIR_TouchConnection = myRoot.Touched:Connect(function(hit)
            if not TATWIR_Active then return end
            local model = hit:FindFirstAncestorOfClass("Model")
            if not model or model == char then return end
            if not model:FindFirstChildOfClass("Humanoid") then return end
            local plr = Players:GetPlayerFromCharacter(model)
            if not plr or plr == LP then return end
            local targetRoot = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso")
            if targetRoot then
                local bv = Instance.new("BodyVelocity")
                bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                bv.Velocity = Vector3.new(0, 250, 0)
                bv.Parent = targetRoot
                task.delay(0.35, function() bv:Destroy() end)
            end
        end)
    end,
    OnDisable = function()
        TATWIR_Active = false
        if TATWIR_SpinConnection then TATWIR_SpinConnection:Disconnect() TATWIR_SpinConnection = nil end
        if TATWIR_TouchConnection then TATWIR_TouchConnection:Disconnect() TATWIR_TouchConnection = nil end
        if TATWIR_BodyAngular then TATWIR_BodyAngular:Destroy() TATWIR_BodyAngular = nil end
    end
})
