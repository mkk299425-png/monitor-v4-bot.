local function Create(instance, parent, props)
    local new = Instance.new(instance)
    if props then
        for prop, value in pairs(props) do
            new[prop] = value
        end
    end
    new.Parent = parent
    return new
end

local function SetProps(instance, props)
    if instance and props then
        for prop, value in pairs(props) do
            instance[prop] = value
        end
    end
    return instance
end

local function Corner(parent, props)
    local new = Create("UICorner", parent)
    new.CornerRadius = Configs_HUB.Corner_Radius
    if props then SetProps(new, props) end
    return new
end

local function Stroke(parent, props)
    local new = Create("UIStroke", parent)
    new.Color = Configs_HUB.Cor_Stroke
    new.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    new.Thickness = 2
    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.25, Color3.fromRGB(180, 180, 180)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(15, 15, 15)),
        ColorSequenceKeypoint.new(0.75, Color3.fromRGB(90, 90, 90)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 255, 255))
    }
    gradient.Rotation = 0
    gradient.Parent = new
    coroutine.wrap(function()
        while new.Parent do
            gradient.Rotation = (gradient.Rotation + 2) % 360
            RunService.RenderStepped:Wait()
        end
    end)()
    if props then SetProps(new, props) end
    return new
end

local function CreateTween(instance, prop, value, time, tweenWait)
    local tween = TweenService:Create(instance,
    TweenInfo.new(time, Enum.EasingStyle.Linear),
    {[prop] = value})
    tween:Play()
    if tweenWait then tween.Completed:Wait() end
end

local function SafeHookFunction(func, newfunc)
    if hookfunction then
        return hookfunction(func, newfunc)
    end
    return func
end

local function FindAllACERemotes()
    local remotes = {}
    local containers = {
        RepStorage, workspace,
        game:GetService("Lighting"),
        game:GetService("CoreGui"),
        game:GetService("StarterGui"),
        LP:FindFirstChild("PlayerGui"),
        LP:FindFirstChild("Backpack"),
        LP:FindFirstChild("PlayerScripts"),
        LP:FindFirstChild("Character") or workspace:FindFirstChild(LP.Name)
    }
    for _, service in ipairs(game:GetChildren()) do
        if service:IsA("Service") and not table.find(containers, service) then
            table.insert(containers, service)
        end
    end
    for _, container in ipairs(containers) do
        if not container then continue end
        local success, descendants = pcall(function() return container:GetDescendants() end)
        if not success then continue end
        for _, obj in ipairs(descendants) do
            if (obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction")) then
                local name = obj.Name
                if string.find(name, "ACE") or string.find(name:lower(), "ace") then
                    table.insert(remotes, obj)
                end
            end
        end
    end
    return remotes
end

local function ApplyBypassToRemote(remote)
    if not remote then return end
    if remote:IsA("RemoteEvent") then
        local oldFire = remote.FireServer
        if oldFire then
            SafeHookFunction(oldFire, function(self, ...)
                local args = {...}
                local msg = tostring(args[1] or "")
                if msg == "HB" then
                    return oldFire(self, unpack(args))
                end
                return nil
            end)
        end
    elseif remote:IsA("RemoteFunction") then
        local oldInvoke = remote.InvokeServer
        if oldInvoke then
            SafeHookFunction(oldInvoke, function(self, ...)
                local args = {...}
                if tostring(args[1]) == "HB" then return "OK" end
                return true
            end)
        end
    end
end

local function stopFly()
    if not Flying then
        FlyPanel.Visible = FlyPanelCreated
        UpdateMainFlyButton()
        return
    end

    Flying = false

    if flightConnection then
        flightConnection:Disconnect()
        flightConnection = nil
    end

    if bodyVelocity then
        bodyVelocity.Velocity = Vector3.zero
        bodyVelocity:Destroy()
        bodyVelocity = nil
    end

    if bodyGyro then
        bodyGyro:Destroy()
        bodyGyro = nil
    end

    if currentHumanoid then
        currentHumanoid.PlatformStand = false
        currentHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        currentHumanoid = nil
    end

    currentRoot = nil
    FlyPanel.Visible = true
    FlyPanelCreated = true
    FlyStopButton.Text = "️ تشغيل الطيران"
    UpdateMainFlyButton()
    MakeNotifi({Title = "Fly", Text = "تم الهبوط — اضغط نفس الزر للطيران مرة ثانية", Time = 2})
end

local function startFly()
    if Flying then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    stopFly()
    currentRoot = root
    currentHumanoid = humanoid

    bodyGyro = Instance.new("BodyGyro", root)
    bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.P = 9e4
    bodyGyro.CFrame = root.CFrame

    bodyVelocity = Instance.new("BodyVelocity", root)
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Velocity = Vector3.zero

    humanoid.PlatformStand = true
    Flying = true
    FlyPanelCreated = true
    FlyPanel.Visible = true
    FlyStopButton.Text = " هبوط"
    UpdateMainFlyButton()
    UpdateFlyPanel()

    local camera = workspace.CurrentCamera

    flightConnection = RunService.Heartbeat:Connect(function()
        if not Flying or not currentRoot or not currentHumanoid or currentHumanoid.Health <= 0 then
            stopFly()
            return
        end

        bodyGyro.CFrame = camera.CFrame

        local moveDir = currentHumanoid.MoveDirection
        if moveDir.Magnitude > 0.05 then
            local camCFrame = camera.CFrame
            local forward = camCFrame.LookVector
            local right = camCFrame.RightVector
            
            local relativeMove = camCFrame:VectorToObjectSpace(moveDir)
            local finalVel = (forward * -relativeMove.Z) + (right * relativeMove.X)
            
            bodyVelocity.Velocity = finalVel * FlySpeed
        else
            bodyVelocity.Velocity = Vector3.zero
        end
    end)

    MakeNotifi({Title = "Fly", Text = "تم تشغيل الطيران", Time = 2})
end

FlyStopButton.Activated:Connect(function()

        stopFly()
    else

    end

        CreateTween(ImageLabel, "ImageColor3", Color3.fromRGB(30, 140, 200), 0.2, true)

            State = newState

local function Monetor_SetCommandStatus(message, isError)
    if Monetor_CommandStatus then
        Monetor_CommandStatus.Text = message or ""
        Monetor_CommandStatus.TextColor3 = isError
            and Color3.fromRGB(255, 120, 120)
            or Color3.fromRGB(180, 255, 180)

local function Monetor_ExecuteTitleCommand(text)
    local commandText = ";titler me " .. text
    local hdClient = RepStorage:WaitForChild("HDAdminHDClient", 5)
    if not hdClient then return false, "HDAdminHDClient غير موجود" end

    local signals = hdClient:WaitForChild("Signals", 5)
    if not signals then return false, "HDAdmin Signals غير موجود" end

    local requestCommand = signals:WaitForChild("RequestCommand", 5)
    if not requestCommand or not requestCommand:IsA("RemoteFunction") then
        return false, "HDAdmin RequestCommand غير متاح"
    end

    local ok, err = pcall(function()
        requestCommand:InvokeServer(commandText)
    end)
    if ok then
        return true, "HDAdminHDClient.Signals.RequestCommand"
    end
    return false, "فشل تنفيذ HDAdmin: " .. tostring(err)
end

    local function SendTitleText()
        if Monetor_CommandBusy then return end

        local value = tostring(Monetor_CommandInput.Text or "")
        value = value:gsub("[%c]", "")
        value = value:match("^%s*(.-)%s*$") or ""

        if value == "" then
            Monetor_SetCommandStatus("اكتب نصًا أولاً", true)
            return
        end

        if #value > 80 then
            value = value:sub(1, 80)
            Monetor_CommandInput.Text = value
        end

        Monetor_CommandBusy = true
        Monetor_SetCommandStatus("جاري الارسال...", false)

        local success, route = Monetor_ExecuteTitleCommand(value)

        if success then
            Monetor_CommandInput.Text = ""
            Monetor_SetCommandStatus("تم الارسال بنجاح", false)
        else
            Monetor_SetCommandStatus(route or "تعذر الوصول إلى HDAdmin", true)
        end

            Monetor_CommandBusy = false
        end)

            SendTitleText()
        end

            planet.Anchored = true
            planet.CanCollide = false
            planet.Material = Enum.Material.Neon
            planet.Color = data.color
            planet.Transparency = 0.15
            planet.CastShadow = false
            planet.Parent = workspace
            
            local light = Instance.new("PointLight")
            light.Color = data.color
            light.Range = 100
            light.Brightness = 2
            light.Parent = planet
            
            table.insert(SkyPlanets, {part = planet, basePos = data.pos, offset = data.pos})
        end
        
        -- تدوير الكواكب
        if SkyRotationConnection then SkyRotationConnection:Disconnect() end
        SkyRotationConnection = RunService.Heartbeat:Connect(function()
            if not CustomSkyEnabled then return end
            local t = tick()
            local cam = workspace.CurrentCamera
            if not cam then return end
            local camPos = cam.CFrame.Position
            
            for i, data in ipairs(SkyPlanets) do
                if data.part and data.part.Parent then
                    -- دوران خفيف + تثبيت حول الكاميرا
                    data.part.CFrame = CFrame.new(camPos + data.basePos + Vector3.new(
                        math.sin(t * 0.3 + i) * 20,
                        math.cos(t * 0.2 + i) * 15,
                        0
                    )) * CFrame.Angles(t * 0.1, t * 0.15 + i, 0)
                end
            end
        end)
        
        MakeNotifi({Title="Sky", Text="السماء الفضائية تفعلت! كواكب + نجوم", Time=4})
    end,
    OnDisable = function()
        CustomSkyEnabled = false
        
        if SkyObject then
            pcall(function() SkyObject:Destroy() end)
            SkyObject = nil
        end
        
        for _, data in ipairs(SkyPlanets) do
            if data.part then pcall(function() data.part:Destroy() end) end
        end
        SkyPlanets = {}
        
        if SkyRotationConnection then
            SkyRotationConnection:Disconnect()
            SkyRotationConnection = nil
        end
        
        -- رجع الإضاءة
        Lighting.Brightness = 2
        Lighting.Ambient = Color3.fromRGB(70, 70, 70)
        Lighting.OutdoorAmbient = Color3.fromRGB(128, 128, 128)
        Lighting.ClockTime = 14
        Lighting.FogEnd = 100000
        
        MakeNotifi({Title="Sky", Text="رجع الجو الطبيعي", Time=2})
    end

                return HttpService:JSONDecode(response)
            end)

                MakeNotifi({Title="Server Hop", Text="❌ فشل جلب السيرفرات", Time=3})
                return
            end
            
            local bestServer = nil
            local bestPlayers = 0
            
            for _, server in ipairs(servers.data) do
                if server.playing < server.maxPlayers and server.id ~= game.JobId then
                    if server.playing > bestPlayers then
                        bestPlayers = server.playing
                        bestServer = server.id
                    end
                end
            end
            
            if bestServer then
                MakeNotifi({Title="Server Hop", Text="✅ وجد سيرفر بـ "..bestPlayers.." لاعب. جاري الانتقال...", Time=3})
                task.wait(1)
                
                pcall(function()
                    game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId, bestServer, LP)
                end)
            else
                MakeNotifi({Title="Server Hop", Text="❌ ما لقيت سيرفر مناسب", Time=3})
            end
        end)
    end
})

local function GetESPColor(player)
    local ok, color = pcall(function()
        if player.Team and player.Team.TeamColor then
            return player.Team.TeamColor.Color
        end
        return Color3.fromRGB(0, 170, 255)
    end)
    return ok and color or Color3.fromRGB(0, 170, 255)
end

local function GetCharacterRoot(character)
    return character and (character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("UpperTorso")
        or character:FindFirstChild("Torso"))
end

local function IsTarget(player)
    return player and ESP_TargetUserId ~= nil and player.UserId == ESP_TargetUserId
end

local function GetOrCreateESPFolder(character)
    if not character then return nil end
    local folder = character:FindFirstChild(ESP_FOLDER_NAME)
    if folder and folder:IsA("Folder") then
        return folder
    end
    if folder then
        pcall(function() folder:Destroy() end)
    end
    folder = Instance.new("Folder")
    folder.Name = ESP_FOLDER_NAME
    folder.Parent = character
    return folder
end

local function RemoveESP(player)
    if not player or not player.Character then return end
    local character = player.Character
    local folder = character:FindFirstChild(ESP_FOLDER_NAME)
    if folder then
        pcall(function() folder:Destroy() end)
    end
end

local function ClearAllESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP then
            RemoveESP(player)
        end
    end
end

local function UpdateNameVisual(player, billboard, distance)
    if not player or not billboard then return end
    local card = billboard:FindFirstChild("Card")
    if not card then return end
    local nameLabel = card:FindFirstChild("PlayerName")
    local infoLabel = card:FindFirstChild("Info")
    local targetBadge = card:FindFirstChild("TargetBadge")
    local humanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
    local target = IsTarget(player)
    local baseColor = target and Color3.fromRGB(255, 55, 55) or GetESPColor(player)

    if nameLabel then
        nameLabel.Text = player.DisplayName ~= "" and player.DisplayName or player.Name
        nameLabel.TextColor3 = target and Color3.fromRGB(255, 85, 85) or Color3.fromRGB(255, 255, 255)
        local accent = nameLabel:FindFirstChild("NameStroke")
        if accent and accent:IsA("UIStroke") then
            accent.Color = baseColor
        end
    end
    if targetBadge then
        targetBadge.Visible = target
    end
    if infoLabel then
        local infoParts = {}
        if ESP_ShowHealth and humanoid then
            table.insert(infoParts, "HP " .. math.max(0, math.floor(humanoid.Health + 0.5)))
        end
        if ESP_ShowDistance and distance and distance < math.huge then
            table.insert(infoParts, math.floor(distance + 0.5) .. "m")
        end
        infoLabel.Text = table.concat(infoParts, "  •  ")
        infoLabel.Visible = (#infoParts > 0)
    end
end

local function GetAutoRefreshHumanoid()
    local character = LP.Character
    if not character then return nil end
    return character:FindFirstChildOfClass("Humanoid")

local function SendRefreshCommand()
    local ok, success, message = pcall(function()
        local hdClient = RepStorage:FindFirstChild("HDAdminHDClient")
        if not hdClient then return false, "HDAdminHDClient غير موجود" end
        local signals = hdClient:FindFirstChild("Signals")
        if not signals then return false, "HDAdmin Signals غير موجود" end
        local requestCommand = signals:FindFirstChild("RequestCommand")
        if not requestCommand or not requestCommand:IsA("RemoteFunction") then
            return false, "RequestCommand غير موجود"
        end
        requestCommand:InvokeServer(";refresh")
        return true, "تم إرسال ;refresh"
    end)
    if not ok then return false, tostring(success) end
    return success == true, message or "تعذر"
end

local function CheckHealthAndRefresh()
    if not AutoRefreshEnabled then return end
    local humanoid = GetAutoRefreshHumanoid()
    if not humanoid or humanoid.MaxHealth <= 0 or humanoid.Health <= 0 then
        AutoRefreshTriggered = false
        return
    end
    local healthPercent = (humanoid.Health / humanoid.MaxHealth) * 100
    if healthPercent <= REFRESH_THRESHOLD then
        if not AutoRefreshTriggered and (os.clock() - LastRefreshTime) >= REFRESH_COOLDOWN then
            AutoRefreshTriggered = true
            LastRefreshTime = os.clock()
            local success, message = SendRefreshCommand()
            MakeNotifi({
                Title = success and "Auto Refresh" or "Auto Refresh — خطأ",
                Text = success and ("تم ;refresh — الدم: " .. math.floor(healthPercent) .. "%") or ("تعذر: " .. tostring(message)),
                Time = success and 2 or 3
            })
        end
    else
        AutoRefreshTriggered = false
    end
end

local function DisconnectAutoRefreshConnections()
    if AutoRefreshConnection then AutoRefreshConnection:Disconnect() AutoRefreshConnection = nil end
    if AutoRefreshCharacterConnection then AutoRefreshCharacterConnection:Disconnect() AutoRefreshCharacterConnection = nil end
    if AutoRefreshHealthConnection then AutoRefreshHealthConnection:Disconnect() AutoRefreshHealthConnection = nil end
end

local function StartAutoRefreshMonitor()
    DisconnectAutoRefreshConnections()
    AutoRefreshTriggered = false
    LastRefreshTime = 0
    AutoRefreshConnection = RunService.RenderStepped:Connect(CheckHealthAndRefresh)
    AutoRefreshCharacterConnection = LP.CharacterAdded:Connect(function(character)
        AutoRefreshTriggered = false
        local humanoid = character:WaitForChild("Humanoid", 5)
        if humanoid and AutoRefreshEnabled then
            if AutoRefreshHealthConnection then AutoRefreshHealthConnection:Disconnect() end
            AutoRefreshHealthConnection = humanoid.HealthChanged:Connect(CheckHealthAndRefresh)
        end
    end)
    local humanoid = GetAutoRefreshHumanoid()
    if humanoid then
        AutoRefreshHealthConnection = humanoid.HealthChanged:Connect(CheckHealthAndRefresh)
    end
end

local function DisconnectAutoRefreshOnDeathConnections()
    if AutoRefreshOnDeathConnection then AutoRefreshOnDeathConnection:Disconnect() AutoRefreshOnDeathConnection = nil end
    if AutoRefreshOnDeathCharConnection then AutoRefreshOnDeathCharConnection:Disconnect() AutoRefreshOnDeathCharConnection = nil end
end

end

local TATWIR_Active = false
local TATWIR_SpinSpeed = 200
local TATWIR_SpeedBox = nil
local TATWIR_SpinConnection = nil

local function TATWIR_Cleanup()
    TATWIR_Active = false
    if TATWIR_SpinConnection then TATWIR_SpinConnection:Disconnect() TATWIR_SpinConnection = nil end
    if TATWIR_TouchConnection then TATWIR_TouchConnection:Disconnect() TATWIR_TouchConnection = nil end
    if TATWIR_BodyAngular then pcall(function() TATWIR_BodyAngular:Destroy() end) TATWIR_BodyAngular = nil end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            for _, part in ipairs(plr.Character:GetDescendants()) do
                if part:IsA("BasePart") then
                    pcall(function()
                        if part.Name == "HumanoidRootPart" then
                            part.CanCollide = false
                        else
                            part.CanCollide = true
                        end
                    end)
                end
            end
        end
    end
end

local function TATWIR_LaunchPlayer(targetChar)
    if not targetChar or targetChar == LP.Character then return end
    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso")
    if not targetRoot then return end
    local oldBV = targetRoot:FindFirstChild("Tatwir_Launch")
    if oldBV then pcall(function() oldBV:Destroy() end) end
    local bv = Instance.new("BodyVelocity")
    bv.Name = "Tatwir_Launch"
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Velocity = Vector3.new(0, 250, 0) + Vector3.new(math.random(-50, 50), 0, math.random(-50, 50))
    bv.Parent = targetRoot
    task.delay(0.35, function() if bv and bv.Parent then bv:Destroy() end end)
end

local function TATWIR_Start()
    if TATWIR_Active then return end
    local char = LP.Character
    if not char then return end
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    TATWIR_Active = true
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            for _, part in ipairs(plr.Character:GetDescendants()) do
                if part:IsA("BasePart") then pcall(function() part.CanCollide = true end) end
            end
        end
    end
    local oldSpin = myRoot:FindFirstChild("Tatwir_Spin")
    if oldSpin then pcall(function() oldSpin:Destroy() end) end
    TATWIR_BodyAngular = Instance.new("BodyAngularVelocity")
    TATWIR_BodyAngular.Name = "Tatwir_Spin"
    TATWIR_BodyAngular.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    TATWIR_BodyAngular.AngularVelocity = Vector3.new(0, math.rad(TATWIR_SpinSpeed), 0)
    TATWIR_BodyAngular.P = 9e9
    TATWIR_BodyAngular.Parent = myRoot
    TATWIR_TouchConnection = myRoot.Touched:Connect(function(hit)
        if not TATWIR_Active then return end
        local model = hit:FindFirstAncestorOfClass("Model")
        if not model or model == char then return end
        if not model:FindFirstChildOfClass("Humanoid") then return end
        local plr = Players:GetPlayerFromCharacter(model)
        if not plr or plr == LP then return end
        TATWIR_LaunchPlayer(model)
    end)
    TATWIR_SpinConnection = RunService.Heartbeat:Connect(function()
        if not TATWIR_Active then return end
        if TATWIR_BodyAngular and TATWIR_BodyAngular.Parent then
            TATWIR_BodyAngular.AngularVelocity = Vector3.new(0, math.rad(TATWIR_SpinSpeed), 0)
        end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                for _, part in ipairs(plr.Character:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide == false and part.Name ~= "HumanoidRootPart" then
                        pcall(function() part.CanCollide = true end)
                    end
                end
            end
        end
    end)
    MakeNotifi({Title="تطوير", Text="التطوير مفعل! لمس أي لاعب = يطير", Time=4})
end

local function TATWIR_Stop()
    if not TATWIR_Active then return end
    TATWIR_Cleanup()
    MakeNotifi({Title="تطوير", Text="تم الإيقاف", Time=2})
end
