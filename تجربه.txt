local function Create(instance, parent, props)
    local new = Instance.new(instance)
    if props then
        for prop, value in pairs(props) do new[prop] = value end
    end
    new.Parent = parent
    return new
end

local function SetProps(instance, props)
    if instance and props then
        for prop, value in pairs(props) do instance[prop] = value end
    end
    return instance
end

local function Corner(parent, props)
    local new = Create("UICorner", parent)
    new.CornerRadius = Configs_HUB.Corner_Radius
    if props then SetProps(new, props) end
    return new
end

local function Stroke(parent, props)
    local new = Create("UIStroke", parent)
    new.Color = Configs_HUB.Cor_Stroke
    new.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    new.Thickness = 1
    if props then SetProps(new, props) end
    return new
end

local function CreateTween(instance, prop, value, time, tweenWait)
    local tween = TweenService:Create(instance,
    TweenInfo.new(time, Enum.EasingStyle.Linear),
    {[prop] = value})
    tween:Play()
    if tweenWait then tween.Completed:Wait() end
end

local function TextSetColor(instance)
    instance.MouseEnter:Connect(function()
        CreateTween(instance, "TextColor3", Color3.fromRGB(180, 180, 180), 0.4, false)
    end)
    instance.MouseLeave:Connect(function()
        CreateTween(instance, "TextColor3", Configs_HUB.Cor_Text, 0.4, false)
    end)
end

local MAX_NOTIFICATIONS = 3

local function NotificationKey(title, text)
    return tostring(title or "Monetor X") .. "\31" .. tostring(text or "")
end

local function RemoveNotification(frame)
    if not frame or not frame.Parent then return end
    for i = #ActiveNotifications, 1, -1 do
        if ActiveNotifications[i].Frame == frame then
            table.remove(ActiveNotifications, i)
            break
        end
    end
    pcall(function() frame:Destroy() end)
end

local BypassEnabled = false

local function SafeHookFunction(func, newfunc)
    if hookfunction then return hookfunction(func, newfunc) end
    return func
end

local function FindAllACERemotes()
    local remotes = {}
    local containers = {
        RepStorage, workspace,
        game:GetService("Lighting"), game:GetService("CoreGui"),
        game:GetService("StarterGui"),
        LP:FindFirstChild("PlayerGui"), LP:FindFirstChild("Backpack"),
        LP:FindFirstChild("PlayerScripts"),
        LP:FindFirstChild("Character") or workspace:FindFirstChild(LP.Name)
    }
    for _, service in ipairs(game:GetChildren()) do
        if service:IsA("Service") and not table.find(containers, service) then
            table.insert(containers, service)
        end
    end
    for _, container in ipairs(containers) do
        if not container then continue end
        local success, descendants = pcall(function() return container:GetDescendants() end)
        if not success then continue end
        for _, obj in ipairs(descendants) do
            if (obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction")) then
                local name = obj.Name
                if string.find(name, "ACE") or string.find(name:lower(), "ace") then
                    table.insert(remotes, obj)
                end
            end
        end
    end
    return remotes
end

local function ApplyBypassToRemote(remote)
    if not remote then return end
    if remote:IsA("RemoteEvent") then
        local oldFire = remote.FireServer
        if oldFire then
            SafeHookFunction(oldFire, function(self, ...)
                local args = {...}
                local msg = tostring(args[1] or "")
                if msg == "HB" then return oldFire(self, unpack(args)) end
                return nil
            end)
        end
    elseif remote:IsA("RemoteFunction") then
        local oldInvoke = remote.InvokeServer
        if oldInvoke then
            SafeHookFunction(oldInvoke, function(self, ...)
                local args = {...}
                if tostring(args[1]) == "HB" then return "OK" end
                return true
            end)
        end
    end
end

local function DisableSprintMovementScans()
    local ps = LP:FindFirstChild("PlayerScripts")
    if not ps then return end
    if not getgc then return end
    local GC = getgc(true)
    for _, obj in ipairs(GC) do
        if type(obj) == "function" then
            local info = debug.getinfo(obj)
            if info and info.what == "Lua" and info.source and string.find(info.source, "SprintMovement") then
                if info.name == "scanForMovementObjects" or info.name == "scanForGuiObjects" then
                    SafeHookFunction(obj, function() return false, nil end)
                end
            end
        end
    end
end

local function DisableLogs()
    if not getconnections then return end
    local conns = getconnections(LogService.MessageOut)
    if conns then
        for _, c in ipairs(conns) do pcall(c.Disconnect, c) end
    end
    pcall(function() 
        if hookfunction then hookfunction(LogService.MessageOut.Connect, function() return nil end) end
    end)
end

local function FullBypass()
    if BypassEnabled then
        MakeNotifi({Title = "Monetor X", Text = "الحماية مكسورة مسبقاً", Time = 2})
        return
    end
    local remotes = FindAllACERemotes()
    if #remotes > 0 then
        for _, r in ipairs(remotes) do ApplyBypassToRemote(r) end
    end
    DisableSprintMovementScans()
    DisableLogs()
    BypassEnabled = true
    MakeNotifi({ Title = "Monetor X", Text = "تم فك الحماية بنجاح", Time = 2 })
end

local function stopFly()
    if not Flying then
        FlyPanel.Visible = FlyPanelCreated
        UpdateMainFlyButton()
        return
    end
    Flying = false
    if flightConnection then flightConnection:Disconnect() flightConnection = nil end
    if bodyVelocity then bodyVelocity.Velocity = Vector3.zero bodyVelocity:Destroy() bodyVelocity = nil end
    if bodyGyro then bodyGyro:Destroy() bodyGyro = nil end
    if currentHumanoid then
        currentHumanoid.PlatformStand = false
        currentHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        currentHumanoid = nil
    end
    currentRoot = nil
    FlyPanel.Visible = true
    FlyPanelCreated = true
    FlyStopButton.Text = "️ تشغيل الطيران"
    UpdateMainFlyButton()
    MakeNotifi({Title = "Fly", Text = "تم الهبوط", Time = 2})
end

local function startFly()
    if Flying then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    stopFly()
    currentRoot = root
    currentHumanoid = humanoid

    bodyGyro = Instance.new("BodyGyro", root)
    bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.P = 9e4
    bodyGyro.CFrame = root.CFrame

    bodyVelocity = Instance.new("BodyVelocity", root)
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Velocity = Vector3.zero

    humanoid.PlatformStand = true
    Flying = true
    FlyPanelCreated = true
    FlyPanel.Visible = true
    FlyStopButton.Text = " هبوط"
    UpdateMainFlyButton()
    UpdateFlyPanel()

    local camera = workspace.CurrentCamera
    flightConnection = RunService.Heartbeat:Connect(function()
        if not Flying or not currentRoot or not currentHumanoid or currentHumanoid.Health <= 0 then
            stopFly() return
        end
        bodyGyro.CFrame = camera.CFrame
        local moveDir = currentHumanoid.MoveDirection
        if moveDir.Magnitude > 0.05 then
            local camCFrame = camera.CFrame
            local forward = camCFrame.LookVector
            local right = camCFrame.RightVector
            local relativeMove = camCFrame:VectorToObjectSpace(moveDir)
            local finalVel = (forward * -relativeMove.Z) + (right * relativeMove.X)
            bodyVelocity.Velocity = finalVel * FlySpeed
        else
            bodyVelocity.Velocity = Vector3.zero
        end
    end)
    MakeNotifi({Title = "Fly", Text = "تم تشغيل الطيران", Time = 2})
end

FlyStopButton.Activated:Connect(function()
    if Flying then stopFly() else startFly() end
end)

function MakeTab(Configs)
    local TabName = Configs.Name or "Tab"
    local TabTitle = Configs.TabTitle or false
    local TabImage = Configs.Image or nil
    local AvatarUsername = Configs.AvatarUsername or nil

    local Frame = Create("Frame", ScrollBar, { Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1 })
    Corner(Frame); Stroke(Frame)
    local TextButton = Create("TextButton", Frame, { Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Text = "" })

    local hasIcon = false

    if AvatarUsername then
        hasIcon = true
        local avatar = Create("ImageLabel", Frame, {
            Size = UDim2.new(0, 20, 0, 20), Position = UDim2.new(0, 5, 0.5, -10),
            BackgroundTransparency = 1, BackgroundColor3 = Configs_HUB.Cor_Options,
            Image = "", Name = "TabAvatar", ZIndex = 5
        })
        Corner(avatar, { CornerRadius = UDim.new(1, 0) })
        local avatarStroke = Instance.new("UIStroke")
        avatarStroke.Thickness = 1.5
        avatarStroke.Color = Color3.fromRGB(255, 255, 255)
        avatarStroke.Transparency = 0.15
        avatarStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        avatarStroke.Parent = avatar
        task.spawn(function()
            local ok, userId = pcall(function() return Players:GetUserIdFromNameAsync(AvatarUsername) end)
            if ok and userId and type(userId) == "number" then
                pcall(function()
                    avatar.Image = "rbxthumb://type=AvatarHeadShot&id=" .. tostring(userId) .. "&w=150&h=150"
                end)
            else
                pcall(function() avatar.Image = "rbxassetid://6031075931" end)
            end
        end)
    elseif TabImage then
        hasIcon = true
        Create("ImageLabel", Frame, {
            Size = UDim2.new(0, 20, 0, 20), Position = UDim2.new(0, 5, 0.5, -10),
            BackgroundTransparency = 1, Image = TabImage, Name = "TabIcon"
        })
    end

    local textOffset = hasIcon and 30 or 5
    local TextLabel = Create("TextLabel", Frame, {
        Size = UDim2.new(1, -textOffset, 1, 0), Position = UDim2.new(0, textOffset, 0, 0),
        BackgroundTransparency = 1, Font = Configs_HUB.Text_Font,
        TextColor3 = textcolor, TextSize = textsize,
        Text = TabName, TextXAlignment = Enum.TextXAlignment.Left
    })

    local Container = Create("ScrollingFrame", Containers, {
        Size = UDim2.new(1, 0, 1, 0), ScrollingDirection = Enum.ScrollingDirection.Y,
        AutomaticCanvasSize = Enum.AutomaticSize.Y, CanvasSize = UDim2.new(),
        BackgroundTransparency = 1, ScrollBarThickness = 2,
        Visible = firstVisible
    })
    Corner(Container); Stroke(Container)
    Create("UIPadding", Container, { PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 10), PaddingTop = UDim.new(0, 10), PaddingBottom = UDim.new(0, 10) })
    Create("UIListLayout", Container, { Padding = UDim.new(0, 5) })

    TextButton.MouseButton1Click:Connect(function()
        for _, container in pairs(Containers:GetChildren()) do
            if container:IsA("ScrollingFrame") then container.Visible = false end
        end
        for _, frame in pairs(ScrollBar:GetChildren()) do
            if frame:IsA("Frame") and frame:FindFirstChild("TextLabel") and frame.TextLabel ~= TextLabel then
                CreateTween(frame.TextLabel, "TextColor3", Configs_HUB.Cor_DarkText, 0.3, false)
                frame.TextLabel.TextSize = 14
            end
        end
        Container.Visible = true
        CreateTween(TextLabel, "TextColor3", Configs_HUB.Cor_Text, 0.3, false)
        TextLabel.TextSize = 15
    end)

    firstVisible = false
    textsize = 14
    textcolor = Configs_HUB.Cor_DarkText
    return Container
end

        ImageColor3 = State and Color3.fromRGB(80, 220, 120) or Configs_HUB.Cor_Stroke
    })

        State = not State
        TextLabel.Text = State and OffText or OnText
        CreateTween(ImageLabel, "ImageColor3", Color3.fromRGB(30, 140, 200), 0.2, true)
        CreateTween(ImageLabel, "ImageColor3", State and Color3.fromRGB(80, 220, 120) or Configs_HUB.Cor_Stroke, 0.2, false)

    end)
    TextSetColor(TextLabel)
    return {
        Button = TextButton, TextLabel = TextLabel,
        GetState = function() return State end,
        SetState = function(newState)
            State = newState
            TextLabel.Text = State and OffText or OnText
            ImageLabel.ImageColor3 = State and Color3.fromRGB(80, 220, 120) or Configs_HUB.Cor_Stroke

    if Monetor_CommandStatus then

local function Monetor_ExecuteTitleCommand(text)
    local commandText = ";titler me " .. text
    local hdClient = RepStorage:WaitForChild("HDAdminHDClient", 5)
    if not hdClient then return false, "HDAdminHDClient غير موجود" end
    local signals = hdClient:WaitForChild("Signals", 5)
    if not signals then return false, "HDAdmin Signals غير موجود" end
    local requestCommand = signals:WaitForChild("RequestCommand", 5)
    if not requestCommand or not requestCommand:IsA("RemoteFunction") then
        return false, "HDAdmin RequestCommand غير متاح"
    end
    local ok, err = pcall(function() requestCommand:InvokeServer(commandText) end)
    if ok then return true, "HDAdmin" end
    return false, "فشل: " .. tostring(err)
end

    return player and ESP_TargetUserId ~= nil and player.UserId == ESP_TargetUserId

    if not character then return nil end

    folder = Instance.new("Folder")

end

local function GetPlayersSortedByDistance()
    local list = {}
    local myCharacter = LP.Character
    local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
    local myPosition = myRoot and myRoot.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP then
            local distance = math.huge
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if myPosition and root then distance = (root.Position - myPosition).Magnitude end
            table.insert(list, {player = player, distance = distance})
        end
    end
    table.sort(list, function(a, b)
        if a.distance == b.distance then return a.player.Name:lower() < b.player.Name:lower() end
        return a.distance < b.distance
    end)
    return list
end

local function UpdateRadarTargetList()
    local options = {}
    for _, entry in ipairs(GetPlayersSortedByDistance()) do
        table.insert(options, entry.player.Name)
    end
    if #options == 0 then options = {"لا يوجد لاعبين"} end
    if ESP_TargetDropdown then
        pcall(function() ESP_TargetDropdown:Update(options) end)
    end
end

local function GetAutoRefreshHumanoid()
    local character = LP.Character
    if not character then return nil end
    return character:FindFirstChildOfClass("Humanoid")
end

local function SendRefreshCommand()
    if RefreshBusy then return false, "قيد التنفيذ" end
    RefreshBusy = true
    local ok, success, message = pcall(function()
        local hdClient = RepStorage:FindFirstChild("HDAdminHDClient")
        if not hdClient then return false, "HDAdminHDClient غير موجود" end
        local signals = hdClient:FindFirstChild("Signals")
        if not signals then return false, "Signals غير موجود" end
        local rc = signals:FindFirstChild("RequestCommand")
        if not rc or not rc:IsA("RemoteFunction") then return false, "RequestCommand غير موجود" end
        rc:InvokeServer(";refresh")
        return true, "تم ;refresh"
    end)
    RefreshBusy = false
    if not ok then return false, tostring(success) end
    return success == true, message
end

local function CheckHealthAndRefresh()
    if not AutoRefreshEnabled then return end
    local humanoid = GetAutoRefreshHumanoid()
    if not humanoid or humanoid.MaxHealth <= 0 or humanoid.Health <= 0 then
        AutoRefreshTriggered = false
        return
    end
    local healthPercent = (humanoid.Health / humanoid.MaxHealth) * 100
    if healthPercent <= REFRESH_THRESHOLD then
        if not AutoRefreshTriggered and (os.clock() - LastRefreshTime) >= REFRESH_COOLDOWN then
            AutoRefreshTriggered = true
            LastRefreshTime = os.clock()
            local success, message = SendRefreshCommand()
            MakeNotifi({
                Title = success and "Auto Refresh" or "خطأ",
                Text = success and ("تم ;refresh — الدم: " .. math.floor(healthPercent) .. "%") or tostring(message),
                Time = 2
            })
        end
    else
        AutoRefreshTriggered = false
    end
end

local function DisconnectAutoRefreshConnections()
    if AutoRefreshConnection then AutoRefreshConnection:Disconnect() AutoRefreshConnection = nil end
    if AutoRefreshCharacterConnection then AutoRefreshCharacterConnection:Disconnect() AutoRefreshCharacterConnection = nil end
    if AutoRefreshHealthConnection then AutoRefreshHealthConnection:Disconnect() AutoRefreshHealthConnection = nil end
end

local function StartAutoRefreshMonitor()
    DisconnectAutoRefreshConnections()
    AutoRefreshTriggered = false
    LastRefreshTime = 0
    AutoRefreshConnection = RunService.RenderStepped:Connect(CheckHealthAndRefresh)
    AutoRefreshCharacterConnection = LP.CharacterAdded:Connect(function(character)
        AutoRefreshTriggered = false
        local humanoid = character:WaitForChild("Humanoid", 5)
        if humanoid and AutoRefreshEnabled then
            if AutoRefreshHealthConnection then AutoRefreshHealthConnection:Disconnect() end
            AutoRefreshHealthConnection = humanoid.HealthChanged:Connect(CheckHealthAndRefresh)
        end
    end)
    local humanoid = GetAutoRefreshHumanoid()
    if humanoid then
        AutoRefreshHealthConnection = humanoid.HealthChanged:Connect(CheckHealthAndRefresh)
    end
end

    if AutoRefreshOnDeathCharConnection then AutoRefreshOnDeathCharConnection:Disconnect() AutoRefreshOnDeathCharConnection = nil end
end

    if AutoRefreshOnDeathConnection then AutoRefreshOnDeathConnection:Disconnect() AutoRefreshOnDeathConnection = nil end

        if not AutoRefreshOnDeathEnabled then return end

            MakeNotifi({
                Title = success and "Death Refresh" or "خطأ",
                Text = success and "تم ;refresh عند الموت" or tostring(message),
                Time = 2
            })
            DeathRefreshBusy = false
        end)
    end)
end

        DeathRefreshBusy = false
        local newHumanoid = character:WaitForChild("Humanoid", 5)

    end)
end

local function TATWIR_Cleanup()
    TATWIR_Active = false
    if TATWIR_SpinConnection then TATWIR_SpinConnection:Disconnect() TATWIR_SpinConnection = nil end
    if TATWIR_TouchConnection then TATWIR_TouchConnection:Disconnect() TATWIR_TouchConnection = nil end
    if TATWIR_BodyAngular then pcall(function() TATWIR_BodyAngular:Destroy() end) TATWIR_BodyAngular = nil end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            for _, part in ipairs(plr.Character:GetDescendants()) do
                if part:IsA("BasePart") then
                    pcall(function()
                        if part.Name == "HumanoidRootPart" then part.CanCollide = false
                        else part.CanCollide = true end
                    end)
                end
            end
        end
    end
    if LP.Character then
        local myRoot = LP.Character:FindFirstChild("HumanoidRootPart")
        if myRoot then
            local oldBV = myRoot:FindFirstChild("Tatwir_Spin")
            if oldBV then pcall(function() oldBV:Destroy() end) end
        end
    end
end

local function TATWIR_LaunchPlayer(targetChar)
    if not targetChar or targetChar == LP.Character then return end
    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso")
    if not targetRoot then return end
    local oldBV = targetRoot:FindFirstChild("Tatwir_Launch")
    if oldBV then pcall(function() oldBV:Destroy() end) end
    local bv = Instance.new("BodyVelocity")
    bv.Name = "Tatwir_Launch"
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Velocity = Vector3.new(0, 250, 0) + Vector3.new(math.random(-50, 50), 0, math.random(-50, 50))
    bv.Parent = targetRoot
    task.delay(0.35, function() if bv and bv.Parent then bv:Destroy() end end)
end

local function TATWIR_Start()
    if TATWIR_Active then return end
    local char = LP.Character
    if not char then return end
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    local myHum = char:FindFirstChildOfClass("Humanoid")
    if not myRoot or not myHum then return end
    TATWIR_Active = true
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            for _, part in ipairs(plr.Character:GetDescendants()) do
                if part:IsA("BasePart") then pcall(function() part.CanCollide = true end) end
            end
        end
    end
    local oldSpin = myRoot:FindFirstChild("Tatwir_Spin")
    if oldSpin then pcall(function() oldSpin:Destroy() end) end
    TATWIR_BodyAngular = Instance.new("BodyAngularVelocity")
    TATWIR_BodyAngular.Name = "Tatwir_Spin"
    TATWIR_BodyAngular.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    TATWIR_BodyAngular.AngularVelocity = Vector3.new(0, math.rad(TATWIR_SpinSpeed), 0)
    TATWIR_BodyAngular.P = 9e9
    TATWIR_BodyAngular.Parent = myRoot
    TATWIR_TouchConnection = myRoot.Touched:Connect(function(hit)
        if not TATWIR_Active then return end
        local model = hit:FindFirstAncestorOfClass("Model")
        if not model or model == char then return end
        if not model:FindFirstChildOfClass("Humanoid") then return end
        local plr = Players:GetPlayerFromCharacter(model)
        if not plr or plr == LP then return end
        TATWIR_LaunchPlayer(model)
    end)
    TATWIR_SpinConnection = RunService.Heartbeat:Connect(function()
        if not TATWIR_Active then return end
        if TATWIR_BodyAngular and TATWIR_BodyAngular.Parent then
            TATWIR_BodyAngular.AngularVelocity = Vector3.new(0, math.rad(TATWIR_SpinSpeed), 0)
        end
    end)
    MakeNotifi({Title = "تطوير", Text = "On", Time = 3})
end

local function TATWIR_Stop()
    if not TATWIR_Active then return end
    TATWIR_Cleanup()
    MakeNotifi({Title="تطوير", Text="Off", Time=2})
end

    if not tool then return false end
    local character = LP.Character
    if not character then return false end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    if tool.Parent == character then return true end
    pcall(function() humanoid:EquipTool(tool) end)
    return true
end

local function GetTargetPlayer()
    if not Target_UserId then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.UserId == Target_UserId then return p end
    end
    return nil
end

local function StopTarget()
    Target_Enabled = false
    Target_Attacking = false
    if Target_Connection then Target_Connection:Disconnect() Target_Connection = nil end
    local ok, tool = HasBanHammer()
    if ok and tool then
        pcall(function()
            if tool:FindFirstChild("Deactivate") then tool.Deactivate(tool) end
        end)
    end
    if LP.Character then
        local hum = LP.Character:FindFirstChildOfClass("Humanoid")
        if hum then hum.WalkSpeed = 16 end
    end
end

local function AttackTwice(tool)
    if not tool then return end
    if Target_Attacking then return end
    Target_Attacking = true
    task.spawn(function()
        pcall(function() if tool:FindFirstChild("Activate") then tool:Activate() end end)
        task.wait(1)
        pcall(function() if tool:FindFirstChild("Activate") then tool:Activate() end end)
        Target_Attacking = false
    end)
end

local function StartTargetLoop()
    if Target_Connection then Target_Connection:Disconnect() end
    Target_Connection = RunService.Heartbeat:Connect(function()
        if not Target_Enabled then return end
        local target = GetTargetPlayer()
        if not target or not target.Character then return end
        local targetRoot = target.Character:FindFirstChild("HumanoidRootPart")
            or target.Character:FindFirstChild("UpperTorso")
            or target.Character:FindFirstChild("Torso")
        if not targetRoot then return end
        local targetHum = target.Character:FindFirstChildOfClass("Humanoid")
        if not targetHum or targetHum.Health <= 0 then
            StopTarget()
            MakeNotifi({Title="Target", Text="مات المستهدف", Time=3})
            if Target_ToggleObj and Target_ToggleObj.SetState then Target_ToggleObj.SetState(false) end
            return
        end
        local myChar = LP.Character
        if not myChar then return end
        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        local myHum = myChar:FindFirstChildOfClass("Humanoid")
        if not myRoot or not myHum or myHum.Health <= 0 then return end
        local targetLook = targetRoot.CFrame.LookVector
        myRoot.CFrame = CFrame.new(targetRoot.Position + targetLook * Target_Offset, targetRoot.Position)
        local hasTool, tool = HasBanHammer()
        if hasTool and tool and tool.Parent ~= myChar then EquipTool(tool) end
        if hasTool and tool and not Target_Attacking then AttackTwice(tool) end
    end)
end

local GraphicsState = {
    Enabled = false,
    Saved = {},
    SavedQuality = nil
}

local function SaveAndDisable(obj)
    if not obj or GraphicsState.Saved[obj] ~= nil then return end
    if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") then
        GraphicsState.Saved[obj] = obj.Enabled
        obj.Enabled = false
    elseif obj:IsA("PointLight") or obj:IsA("SpotLight") or obj:IsA("SurfaceLight") then
        GraphicsState.Saved[obj] = obj.Enabled
        obj.Enabled = false
    elseif obj:IsA("PostEffect") then
        GraphicsState.Saved[obj] = obj.Enabled
        obj.Enabled = false
    end
end

local function ApplyPerformanceMode()
    if GraphicsState.Enabled then return end
    GraphicsState.Enabled = true
    GraphicsState.Saved = {}

    pcall(function()
        local settings = UserSettings():GetService("UserGameSettings")
        GraphicsState.SavedQuality = settings.GraphicsQualityLevel
        settings.GraphicsQualityLevel = Enum.SavedQualitySetting.QualityLevel1
    end)

    pcall(function()
        GraphicsState.Saved.GlobalShadows = game:GetService("Lighting").GlobalShadows
        game:GetService("Lighting").GlobalShadows = false
        if workspace:FindFirstChildOfClass("Terrain") then
            local terrain = workspace:FindFirstChildOfClass("Terrain")
            GraphicsState.Saved.TerrainDecoration = terrain.Decoration
            terrain.Decoration = false
        end
    end)

    pcall(function()
        for _, obj in ipairs(game:GetDescendants()) do
            SaveAndDisable(obj)
        end
    end)

    MakeNotifi({Title="الجرافكس", Text="تم تفعيل وضع الأداء لتقليل الحمل على الهاتف", Time=2})
end

local function RestoreGraphics()
    if not GraphicsState.Enabled then return end
    GraphicsState.Enabled = false

    pcall(function()
        local settings = UserSettings():GetService("UserGameSettings")
        if GraphicsState.SavedQuality then
            settings.GraphicsQualityLevel = GraphicsState.SavedQuality
        end
    end)

    pcall(function()
        local lighting = game:GetService("Lighting")
        if GraphicsState.Saved.GlobalShadows ~= nil then
            lighting.GlobalShadows = GraphicsState.Saved.GlobalShadows
        end
        local terrain = workspace:FindFirstChildOfClass("Terrain")
        if terrain and GraphicsState.Saved.TerrainDecoration ~= nil then
            terrain.Decoration = GraphicsState.Saved.TerrainDecoration
        end
    end)

    pcall(function()
        for obj, oldValue in pairs(GraphicsState.Saved) do
            if typeof(obj) == "Instance" and obj.Parent then
                local isSupported =
                    obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or
                    obj:IsA("PointLight") or obj:IsA("SpotLight") or obj:IsA("SurfaceLight") or
                    obj:IsA("PostEffect")
                if isSupported then
                    obj.Enabled = oldValue
                end
            end
        end
    end)

    GraphicsState.Saved = {}
    MakeNotifi({Title="الجرافكس", Text="تم إيقاف وضع الأداء وإرجاع الإعدادات", Time=2})
end
